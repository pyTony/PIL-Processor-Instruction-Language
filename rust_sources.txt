//! Decoded 386 addresses and already-loaded segment caches. No byte decoder.
use super::{ExecError, MemWidth};

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum Gpr { Eax, Ecx, Edx, Ebx, Esp, Ebp, Esi, Edi }
impl Gpr {
    fn index(self) -> usize { self as usize }
}
#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum SegmentRegister { Es, Cs, Ss, Ds, Fs, Gs }
impl SegmentRegister {
    pub fn index(self) -> usize { self as usize }
}
#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum Address16 { BxSi, BxDi, BpSi, BpDi, Si, Di, Bp, Bx, Direct }
#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum EffectiveAddress {
    Bits16 { form: Address16, displacement: u16 },
    Bits32 { base: Option<Gpr>, index: Option<Gpr>, scale: u8, displacement: u32 },
}
#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct ResolvedAddress { pub offset: u32, pub segment: SegmentRegister }
impl EffectiveAddress {
    // Displacements have already been sign-extended (disp8) and normalized to
    // address size by the caller. Address size is independent of operand size.
    pub fn resolve(self, regs: &[u32;8], override_segment: Option<SegmentRegister>) -> Result<ResolvedAddress, ExecError> {
        let (offset, stack) = match self {
            Self::Bits16 { form, displacement } => {
                let (base, index) = match form {
                    Address16::BxSi => (Some(Gpr::Ebx), Some(Gpr::Esi)),
                    Address16::BxDi => (Some(Gpr::Ebx), Some(Gpr::Edi)),
                    Address16::BpSi => (Some(Gpr::Ebp), Some(Gpr::Esi)),
                    Address16::BpDi => (Some(Gpr::Ebp), Some(Gpr::Edi)),
                    Address16::Si => (None, Some(Gpr::Esi)),
                    Address16::Di => (None, Some(Gpr::Edi)),
                    Address16::Bp => (Some(Gpr::Ebp), None),
                    Address16::Bx => (Some(Gpr::Ebx), None),
                    Address16::Direct => (None, None),
                };
                let sum = base.map_or(0, |r|regs[r.index()]).wrapping_add(index.map_or(0, |r|regs[r.index()])).wrapping_add(u32::from(displacement));
                (sum & 0xffff, base == Some(Gpr::Ebp))
            }
            Self::Bits32 { base, index, scale, displacement } => {
                if ![1,2,4,8].contains(&scale) || index == Some(Gpr::Esp) || (index.is_none() && scale != 1) {
                    return Err(ExecError::InvalidContext);
                }
                let offset = base.map_or(0, |r|regs[r.index()]).wrapping_add(index.map_or(0, |r|regs[r.index()]).wrapping_mul(u32::from(scale))).wrapping_add(displacement);
                (offset, matches!(base, Some(Gpr::Ebp | Gpr::Esp)))
            }
        };
        Ok(ResolvedAddress { offset, segment: override_segment.unwrap_or(if stack {SegmentRegister::Ss} else {SegmentRegister::Ds}) })
    }
}

// These are normalized hidden-cache attributes, not raw GDT/LDT descriptors.
// Caller has already performed descriptor-load, present and privilege checks.
#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum SegmentAccess {
    // Supplied by callers/tests; the current monitor creates usable caches only.
    #[allow(dead_code)]
    Unusable,
    Data { writable: bool, expand_down: bool, big: bool },
    Code { readable: bool },
}
#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct SegmentCache { pub base: u32, pub limit: u32, pub access: SegmentAccess }
impl SegmentCache {
    // Ordinary real-mode cache after a segment load; does not model transitions
    // from protected mode with retained hidden caches ("unreal mode").
    // CS uses these ordinary real-mode data-access permissions as well;
    // Code caches describe protected-mode permissions, not the register name.
    pub fn real_mode(selector: u16) -> Self {
        Self { base: u32::from(selector) << 4, limit: 0xffff,
            access: SegmentAccess::Data { writable: true, expand_down: false, big: false } }
    }
}
#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct MemoryContext { pub segment: SegmentRegister, pub cache: SegmentCache }
impl MemoryContext {
    pub fn translate(self, offset: u32, width: MemWidth, write: bool, eip: u32) -> Result<u32, ExecError> {
        self.translate_span(u64::from(offset),u32::from(width.bytes()),write,eip)
    }
    // Widened offsets let multi-slot stack frames reject wrap before any reads.
    pub fn translate_span(self, offset:u64, bytes:u32, write:bool, eip:u32)->Result<u32,ExecError>{
        // An invalid loaded SS cache is a caller error: loading SS is not yet
        // implemented and would have rejected null/code/read-only descriptors.
        if self.segment == SegmentRegister::Ss && !matches!(self.cache.access, SegmentAccess::Data { writable: true, .. }) {
            return Err(ExecError::InvalidContext);
        }
        let gp = ExecError::GeneralProtection { eip, error_code: 0 };
        let (permitted, expand_down, big) = match self.cache.access {
            SegmentAccess::Unusable => return Err(gp),
            SegmentAccess::Data { writable, expand_down, big } => (!write || writable, expand_down, big),
            SegmentAccess::Code { readable } => (!write && readable, false, false),
        };
        if !permitted { return Err(gp); }
        // Use a widened end: a multi-byte operand cannot wrap its segment offset.
        let end = offset.checked_add(u64::from(bytes).saturating_sub(1)).unwrap_or(u64::MAX);
        let in_bounds = if expand_down {
            offset > u64::from(self.cache.limit) && end <= if big {u64::from(u32::MAX)} else {0xffff}
        } else { end <= u64::from(self.cache.limit) };
        if bytes==0 || offset>u64::from(u32::MAX) || !in_bounds {
            return Err(if self.segment == SegmentRegister::Ss {
                ExecError::StackSegment { eip, error_code: 0 }
            } else { gp });
        }
        Ok(self.cache.base.wrapping_add(offset as u32))
    }
}
#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct AddressContext {
    pub effective: EffectiveAddress,
    pub override_segment: Option<SegmentRegister>,
    // ES, CS, SS, DS, FS, GS, in SegmentRegister order.
    pub segments: [SegmentCache;6],
}
impl AddressContext {
    pub fn resolve(self, regs: &[u32;8]) -> Result<(u32, MemoryContext), ExecError> {
        let address = self.effective.resolve(regs, self.override_segment)?;
        Ok((address.offset, MemoryContext { segment: address.segment, cache: self.segments[address.segment.index()] }))
    }
}

// Mutable architectural segment state. Only ordinary real-mode loads are admitted.
#[derive(Clone,Copy,Debug,PartialEq,Eq)]
pub struct SegmentState {
    pub segments:[SegmentCache;6],pub selectors:[u16;6],
    pub msw:u16,pub real_mode:bool,pub ss_shadow:u8,pub sti_shadow:u8,pub flag_checkpoint:Option<(u32,u32,u32)>,
}
impl SegmentState {
    // The mode bit is coherent with the existing normalized mode state.
    pub fn machine_status(&self)->u16{(self.msw&!1)|u16::from(!self.real_mode)}
    pub fn checkpoint_flags(&mut self,eip:u32,flags:u32,undefined:u32){if self.flag_checkpoint.map(|v|v.0)!=Some(eip){self.flag_checkpoint=Some((eip,flags,undefined));}}
    pub fn recover_flags(&mut self)->Option<(u32,u32)>{self.flag_checkpoint.take().map(|(_,f,u)|(f,u))}
    pub fn retired(&mut self){self.flag_checkpoint=None;self.ss_shadow=self.ss_shadow.saturating_sub(1);self.sti_shadow=self.sti_shadow.saturating_sub(1);}
}

//! Bounded decoded-instruction/PIL cache. Guest execution still interprets tokens.
use std::collections::BTreeMap;
use std::rc::Rc;
use super::guest::{Decoded,GuestError,Image};
use super::memory::Ram;

pub const CAPACITY:usize=256;
#[derive(Clone,Copy,Debug,Default,PartialEq,Eq)]
pub struct Stats {pub hits:u64,pub misses:u64,pub invalidations:u64,pub evictions:u64}
struct Entry {image:Image,signature:[u8;15],decoded:Rc<Decoded>}
pub struct DecodeCache {enabled:bool,entries:BTreeMap<u32,Entry>,stats:Stats}
impl Default for DecodeCache {
    fn default()->Self {Self{enabled:true,entries:BTreeMap::new(),stats:Stats::default()}}
}
impl DecodeCache {
    pub fn enabled(&self)->bool {self.enabled}
    pub fn stats(&self)->Stats {self.stats}
    pub fn clear(&mut self) {self.entries.clear();self.stats=Stats::default();}
    pub fn set_enabled(&mut self,enabled:bool) {self.enabled=enabled;self.clear();}
    pub fn summary(&self)->String {
        format!("cache={} entries={}/{} hits={} misses={} invalidations={} evictions={}",if self.enabled{"on"}else{"off"},self.entries.len(),CAPACITY,self.stats.hits,self.stats.misses,self.stats.invalidations,self.stats.evictions)
    }
    pub fn fetch(&mut self,image:&Image,ram:&Ram,eip:u32)->Result<Rc<Decoded>,GuestError> {
        if !self.enabled {return image.decode(ram,eip).map(Rc::new);}
        if let Some(entry)=self.entries.get(&eip) {
            if entry.image==*image {
                // Re-fetch every encoded byte through the current fetch checks.
                // This catches poke, guest stores, overlap and changed lengths
                // without requiring every future RAM writer to notify the cache.
                match image.fetch_signature(ram,eip,entry.decoded.length) {
                    Ok(signature) if signature==entry.signature=>{
                        self.stats.hits+=1;return Ok(Rc::clone(&entry.decoded));
                    }
                    Err(error)=>{
                        self.entries.remove(&eip);self.stats.invalidations+=1;return Err(error);
                    }
                    _=>{},
                }
            }
            self.entries.remove(&eip);self.stats.invalidations+=1;
        }
        self.stats.misses+=1;
        let decoded=Rc::new(image.decode(ram,eip)?);
        let signature=image.fetch_signature(ram,eip,decoded.length)?;
        if self.entries.len()==CAPACITY {
            self.stats.evictions+=self.entries.len() as u64;self.entries.clear();
        }
        self.entries.insert(eip,Entry{image:image.clone(),signature,decoded:Rc::clone(&decoded)});
        Ok(decoded)
    }
}

fn zip(data:&[u8])->Result<Archive,String>{

    let eocd=(data.len().saturating_sub(65557)..=data.len()-22).rev().find(|&n|data[n..].starts_with(b"PK\x05\x06")&&u16(data,n+20).is_ok_and(|c|n+22+usize::from(c)==data.len())).ok_or("ZIP end directory missing")?;
    if u16(data,eocd+4)?!=0||u16(data,eocd+6)?!=0||u16(data,eocd+8)?!=u16(data,eocd+10)?{return Err("multi-disk ZIP is not supported".into());}

    if count>=65535||count>MAX_FILES||start.checked_add(length)!=Some(eocd){return Err("ZIP64/nonstandard directory layout is not supported".into());}
    let mut at=start;let mut archive=Archive::new();
    for _ in 0..count{
        if !range(data,at,46)?.starts_with(b"PK\x01\x02"){return Err("invalid ZIP central record".into());}
        let flags=u16(data,at+8)?;let method=u16(data,at+10)?;let crc=u32(data,at+16)?;
        let compressed=u32(data,at+20)?as usize;let size=u32(data,at+24)?as usize;
        let n=usize::from(u16(data,at+28)?);let extra=usize::from(u16(data,at+30)?);let comment=usize::from(u16(data,at+32)?);
        if flags&1!=0||flags&0x40!=0{return Err("encrypted ZIP is not supported".into());}
        if u16(data,at+34)?!=0||compressed==u32::MAX as usize||size==u32::MAX as usize{return Err("ZIP64/multi-disk entry is not supported".into());}
        if (u32(data,at+38)?>>16)&0xf000==0xa000{return Err("ZIP symbolic links are not extracted".into());}
        let namebytes=range(data,at+46,n)?;
        if flags&0x800==0&&!namebytes.is_ascii(){return Err("non-ASCII legacy ZIP filenames are not supported".into());}
        let name=std::str::from_utf8(namebytes).map_err(|_|"invalid ZIP UTF8 filename")?;
        let local=u32(data,at+42)?as usize;at=at.checked_add(46+n+extra+comment).ok_or("ZIP directory overflow")?;
        if at>eocd{return Err("ZIP central directory crosses boundary".into());}
        if name.ends_with('/')||name.ends_with('\\'){safe_name(name.trim_end_matches(|c|c=='/'||c=='\\'))?;continue;}
        let name=safe_name(name)?;
        if size>MAX_ENTRY||archive.total+size>MAX_TOTAL{return Err("ZIP expansion limit exceeded".into());}
        if !range(data,local,30)?.starts_with(b"PK\x03\x04")||u16(data,local+6)?!=flags||u16(data,local+8)?!=method{return Err("ZIP local/central header mismatch".into());}
        let ln=usize::from(u16(data,local+26)?);let le=usize::from(u16(data,local+28)?);
        if range(data,local+30,ln)?!=namebytes{return Err("ZIP local/central filename mismatch".into());}
        if flags&8==0&&(u32(data,local+14)?!=crc||u32(data,local+18)?as usize!=compressed||u32(data,local+22)?as usize!=size){return Err("ZIP local/central size or CRC mismatch".into());}
        let payload=local.checked_add(30+ln+le).ok_or("ZIP local offset overflow")?;
        if payload.checked_add(compressed).is_none_or(|end|end>start){return Err("ZIP payload crosses directory boundary".into());}
        let input=range(data,payload,compressed)?;

        if crc32(&bytes)!=crc{return Err(format!("ZIP CRC32 mismatch: {name}"));}archive.add(name,bytes)?;
    }
    if at!=eocd{return Err("ZIP central directory size mismatch".into());}Ok(archive)
}
struct Fat<'a>{data:&'a[u8],fat:&'a[u8],cluster_base:usize,unit:usize,clusters:usize,dirs:BTreeSet<usize>,archive:Archive}

pub fn parse(data:&[u8],is_zip:bool)->Result<Archive,String>{
    if data.len()>MAX_INPUT{return Err("container exceeds64MiB input limit".into());}
    let mut archive=if is_zip{zip(data)?}else{fat16(data)?};
    // Exactly one nested package level. Keep original ZIP bytes as well.
    let originals=archive.files.len();
    for i in 0..originals{if !archive.files[i].path.to_ascii_lowercase().ends_with(".zip"){continue;}

            Ok(nested)=>{let prefix=format!("{}.contents/",archive.files[i].path);for e in nested.files{archive.add(format!("{prefix}{}",e.path),e.bytes)?;}},
            Err(e)=>archive.notes.push(format!("ZIP package not expanded:{}:{e}",archive.files[i].path)),
        }
    }Ok(archive)
}

    Cpu { program: Program, dst: View, src: View, context: FlowContext },

    fn step_image(&mut self)->Stop {
        use super::guest::{GuestError,Operation};
        let decoded=match self.cache.fetch(self.image.as_ref().unwrap(),&self.ram,self.cpu.eip) {
            Ok(d)=>d,Err(GuestError::Execution(e))=>return Stop::Fault(e),
            Err(GuestError::Decode(message))=>return Stop::Decode{eip:self.cpu.eip,message},
        };
        match &decoded.operation {
            Operation::Pil{program,dst,src,flow,address}=>{
                let mut machine=self.image.as_ref().unwrap().segment_state();
                let result=super::execute_machine(program,&mut self.cpu,*dst,*src,*flow,&mut self.ram,*address,&mut machine);
                self.image.as_mut().unwrap().set_segment_state(machine);
                match result{Ok(())=>Stop::Advanced,Err(e)=>Stop::Fault(e)}
            }
            Operation::Input{next}=>{
                let byte=match self.serial.try_read(){Some(b)=>b,None=>return Stop::WaitingRx(self.cpu.eip)};
                self.cpu.regs[0]=(self.cpu.regs[0]&!255)|u32::from(byte);self.cpu.eip = *next;if let Some(image)=self.image.as_mut(){image.retired();}Stop::Advanced
            }
            Operation::Output{next}=>{
                if self.serial.try_write(self.cpu.regs[0] as u8).is_err(){return Stop::TxBlocked(self.cpu.eip);}
                self.tx_events.push(self.cpu.regs[0] as u8);self.cpu.eip = *next;if let Some(image)=self.image.as_mut(){image.retired();}Stop::Advanced
            }
            Operation::Halt=>Stop::Halt(self.cpu.eip),
        }
    }
    fn reset(&mut self) {
        self.cache.clear();
        self.image=self.initial_image.clone();
        self.cpu=self.initial.clone();self.ram=self.initial_ram.clone();self.serial.clear();self.tx_events.clear();self.stopped_at_break=None;
        // Keep breakpoints while resetting the currently loaded demo.
    }
    fn step(&mut self) -> Stop {
        self.stopped_at_break=None;
        if self.image.is_some(){return self.step_image();}
        let instruction=match self.instructions.get(&self.cpu.eip) {
            Some(i)=>i, None=>return Stop::MissingInstruction(self.cpu.eip),
        };
        match &instruction.action {
            Action::Cpu{program,dst,src,context} => match execute_flow(program,&mut self.cpu,*dst,*src,*context) {
                Ok(())=>Stop::Advanced,Err(e)=>Stop::Fault(e),
            },
            Action::CpuMemory{program,dst,src,context,address}=>match execute_memory(program,&mut self.cpu,*dst,*src,*context,&mut self.ram,*address) {
                Ok(())=>Stop::Advanced,Err(e)=>Stop::Fault(e),
            },
            Action::MockRx{next} => {
                let byte=match self.serial.try_read() {Some(b)=>b,None=>return Stop::WaitingRx(self.cpu.eip)};
                self.cpu.regs[0]=(self.cpu.regs[0]&!255)|u32::from(byte);
                self.cpu.eip = *next;if let Some(image)=self.image.as_mut(){image.retired();}Stop::Advanced
            }
            Action::MockTx{next} => {
                if self.serial.try_write(self.cpu.regs[0] as u8).is_err() {
                    return Stop::TxBlocked(self.cpu.eip);
                }
                self.tx_events.push(self.cpu.regs[0] as u8);self.cpu.eip = *next;if let Some(image)=self.image.as_mut(){image.retired();}Stop::Advanced
            }
            Action::HarnessHalt=>Stop::Halt(self.cpu.eip),
        }
    }
    fn run(&mut self, budget:usize)->Result<(usize,Stop),String> {
        if budget==0 || budget>MAX_RUN {return Err(format!("run budget must be 1..{MAX_RUN}"));}
        let resume=self.stopped_at_break.take().filter(|eip|*eip==self.cpu.eip);
        for done in 0..budget {
            if self.breakpoints.contains(&self.cpu.eip) && !(done==0 && resume==Some(self.cpu.eip)) {
                self.stopped_at_break=Some(self.cpu.eip);
                return Ok((done,Stop::Breakpoint(self.cpu.eip)));
            }
            let stop=self.step();
            if stop!=Stop::Advanced {return Ok((done,stop));}
        }
        Ok((budget,Stop::Budget(self.cpu.eip)))
    }
    fn run_large(&mut self,budget:usize)->Result<(usize,Stop),String> {
        if budget==0 || budget>MAX_TIMED{return Err(format!("timed budget must be 1..{MAX_TIMED}"));}
        let mut done=0;
        loop {
            let (n,stop)=self.run((budget-done).min(MAX_RUN))?;done+=n;
            match stop {
                Stop::Budget(_) if done<budget=>{},
                other=>return Ok((done,other)),
            }
        }
    }
    fn timed_run(&mut self,budget:usize)->Result<String,String> {
        // Validate before taking/altering any debugger continuation state.
        if budget==0 || budget>MAX_TIMED{return Err(format!("timed budget must be 1..{MAX_TIMED}"));}
        let start=std::time::Instant::now();let (n,stop)=self.run_large(budget)?;let elapsed=start.elapsed();
        let rate=if n==0 || elapsed.as_nanos()==0 {"rate unavailable (no completed steps or clock interval)".into()}
            else {format!("{:.0} guest instructions/s; {:.1} ns/instruction",n as f64/elapsed.as_secs_f64(),elapsed.as_secs_f64()*1e9/n as f64)};
        Ok(format!("{n} completed steps; {}\nelapsed={:.3} ms; {rate}\nHost wall time: PIL token interpreter; fetch validation, cache misses/compilation and dispatch; not 386 cycles.\n{}",Self::stop_text(&stop),elapsed.as_secs_f64()*1000.0,self.cache.summary()))
    }

        }

        }

        if self.tx_events!=[0x41] {return Err("selftest expects exactly TX=41; queue RX=41 before running".into());}
        Ok(())
    }

    }

    fn check_current(&mut self)->Result<String,String> {

        }

    fn listing(&self,start:u32,count:u64)->Result<String,String>{
        if count==0||count>256{return Err("list count must be 1..256 instructions".into());}
        let mut lines=Vec::new();
        if let Some(image)=&self.image{
            if !image.contains_offset(start){return Err("listing address outside loaded image".into());}
            let mut ip=start;
            for _ in 0..count {
                if !image.contains_offset(ip){break;}
                match image.decode(&self.ram,ip){
                    Ok(d)=>{
                        let signature=image.fetch_signature(&self.ram,ip,d.length).map_err(|e|format!("{e:?}"))?;
                        let bytes=signature[..d.length as usize].iter().map(|b|format!("{b:02x}")).collect::<Vec<_>>().join(" ");
                        lines.push(format!("{} {ip:08x}  {bytes:<44} {}",if ip==self.cpu.eip{"*"}else{" "},d.label));
                        ip+=d.length;if matches!(d.operation,super::guest::Operation::Halt){break;}
                    }
                    Err(e)=>{lines.push(format!("  {ip:08x}  {e:?}"));break;}
                }
            }
        }else{
            if !self.instructions.contains_key(&start){return Err("no instruction at listing address".into());}
            for (ip,instruction) in self.instructions.range(start..).take(count as usize){lines.push(format!("{} {ip:08x} {}",if *ip==self.cpu.eip{"*"}else{" "},instruction.label));}
        }
        Ok(lines.join("\n"))
    }
    fn dump_memory(&self,address:u32,count:u64,width:super::memory::MemWidth,decimal:bool)->Result<String,String>{
        if count==0||count>256{return Err("dump count must be 1..256 values".into());}
        let bytes=usize::from(width.bytes());let length=(count as usize)*bytes;
        if u64::from(address)+length as u64>0x1_0000_0000{return Err("RAM range out of bounds".into());}
        let start=usize::try_from(address).map_err(|_|"address out of range")?;
        let end=start.checked_add(length).ok_or("address out of range")?;
        let data=self.ram.bytes().get(start..end).ok_or("RAM range out of bounds")?;
        let digits=if decimal{match bytes{1=>3,2=>5,_=>10}}else{bytes*2};
        let columns=(16/bytes)*(digits+1)-1;let mut lines=Vec::new();
        for (row,chunk) in data.chunks(16).enumerate(){
            let values=chunk.chunks_exact(bytes).map(|item|{
                let value=item.iter().enumerate().fold(0u32,|value,(i,b)|value|(u32::from(*b)<<(i*8)));
                if decimal{format!("{value:>digits$}")}else{format!("{value:0digits$x}")}
            }).collect::<Vec<_>>().join(" ");
            let ascii:String=chunk.iter().map(|&b|if (0x20..=0x7e).contains(&b){char::from(b)}else{'.'}).collect();
            let at=u64::from(address)+(row*16) as u64;
            lines.push(format!("{at:08x}: {values:<columns$} |{ascii:<16}|"));
        }
        Ok(lines.join("\n"))
    }

        for b in &case.tx {d.serial.try_write(*b).map_err(|e|format!("{e:?}"))?;}
        let mut stopped=false;
        for _ in 0..1000 {
            let stop=d.step();let status=match stop {

                Stop::WaitingRx(ip)=>[2,ip,0,0,0],Stop::TxBlocked(ip)=>[3,ip,0,0,0],

            };

fn bytes_at(p:&Loaded,n:u32)->Result<Vec<u8>,String>{
    p.image.fetch_signature(&p.ram,p.cpu.eip,n).map(|b|b[..n as usize].to_vec()).map_err(|e|format!("{e:?}"))

fn site_bytes(p:&Loaded)->Vec<u8>{

fn opcode_key(bytes:&[u8])->String{
    let at=bytes.iter().position(|b|!matches!(*b,0x26|0x2e|0x36|0x3e|0x64|0x65|0x66|0x67)).unwrap_or(0);
    match bytes.get(at){Some(0x0f)=>format!("opcode 0f {}",bytes.get(at+1).map(|x|format!("{x:02x}")).unwrap_or_else(||"??".into())),Some(b)=>format!("opcode {b:02x}"),None=>"fetch/truncated encoding".into()}
}
fn service(p:&mut Loaded,vector:u8,out:&mut Vec<u8>)->Result<Option<u8>,String>{
    match(vector,(p.cpu.regs[0]>>8)as u8){
        (0x20,_)=>if p.selectors[SegmentRegister::Cs.index()]==PSP{Ok(Some(0))}else{Err("INT20 requires CS=PSP in this shim".into())},
        (0x21,0x02)=>{
            let byte=p.cpu.regs[2]as u8;
            // Derive the bounded capture's column, including earlier AH09 text.
            // LF preserves column; CR resets it. This is not a text-screen device.
            let count=if byte==9{
                let mut column=0usize;
                for &b in out.iter(){match b{b'\r'=>column=0,b'\n'|7=>{},8=>column=column.saturating_sub(1),9=>column+=8-column%8,_=>column+=1}}
                8-column%8
            }else{1};
            if out.len().checked_add(count).is_none_or(|n|n>MAX_OUTPUT){return Err("DOS output limit reached before character output".into());}
            let value=if byte==9{b' '}else{byte};
            out.extend(std::iter::repeat(value).take(count));
            p.cpu.regs[0]=(p.cpu.regs[0]&!255)|u32::from(value);
            Ok(None)
        },
        (0x21,0x37)=>{
            if p.cpu.regs[0]as u8!=0{return Err("DOS switch-character shim supports AH37/AL00 query only".into());}
            p.cpu.regs[0]&=!255;p.cpu.regs[2]=(p.cpu.regs[2]&!255)|u32::from(b'/');
            Ok(None)
        },
        (0x21,0x25)=>{
            // AL selects a physical IVT entry; DS:DX is a pointer value, not
            // an access to the handler. One dword store checks all four bytes.
            let address=(p.cpu.regs[0]&255)*4;
            let pointer=(u32::from(p.image.selectors[SegmentRegister::Ds.index()])<<16)|(p.cpu.regs[2]&0xffff);
            p.ram.store(address,MemWidth::Dword,pointer).map_err(|e|format!("DOS vector write: {e:?}"))?;
            Ok(None)
        },
        (0x21,0x35)=>{
            // Read the complete pointer before changing BX or the ES cache.
            let address=(p.cpu.regs[0]&255)*4;
            let pointer=p.ram.load(address,MemWidth::Dword).map_err(|e|format!("DOS vector read: {e:?}"))?;
            let selector=(pointer>>16)as u16;
            p.cpu.regs[3]=(p.cpu.regs[3]&0xffff0000)|(pointer&0xffff);
            let es=SegmentRegister::Es.index();
            p.image.selectors[es]=selector;p.selectors[es]=selector;
            p.image.segments[es]=SegmentCache::real_mode(selector);
            Ok(None)
        },
        (0x21,0x30)=>{
            // Read before changing registers so a failed PSP read has no effects.
            let version=p.ram.load(u32::from(PSP)*16+0x40,MemWidth::Word).map_err(|e|format!("DOS version PSP read: {e:?}"))?;
            let flags=p.cpu.regs[0]as u8==1;
            p.cpu.regs[0]=(p.cpu.regs[0]&0xffff0000)|version;
            p.cpu.regs[3]=(p.cpu.regs[3]&0xffff0000)|if flags{0}else{0xff00};
            p.cpu.regs[1]&=0xffff0000;
            Ok(None)
        },
        (0x21,0x4c)=>Ok(Some(p.cpu.regs[0]as u8)),
        (0x21,ah @ (0x48|0x49|0x4a))=>{memory_service(p,ah)?;Ok(None)},
        (0x21,9)=>{
            let offset=p.cpu.regs[2]&0xffff;let mut text=Vec::new();
            let context=MemoryContext{segment:SegmentRegister::Ds,cache:p.image.segments[SegmentRegister::Ds.index()]};
            for n in offset..=0xffff{
                let addr=context.translate(n,MemWidth::Byte,false,p.cpu.eip).map_err(|e|format!("DOS string access: {e:?}"))?;
                let b=p.ram.load(addr,MemWidth::Byte).map_err(|e|format!("DOS string RAM access: {e:?}"))?as u8;
                if b==b'$'{
                    out.extend(text);p.cpu.regs[0]=(p.cpu.regs[0]&!255)|0x24;return Ok(None);
                }
                if out.len()+text.len()>=MAX_OUTPUT{return Err("DOS output limit reached before string terminator".into());}
                text.push(b);
            }Err("DOS string has no $ terminator before segment end".into())
        },_=>Err(format!("unimplemented DOS/BIOS service INT {vector:02x}, AH={:02x}",(p.cpu.regs[0]>>8)&255)),
    }
}
pub fn run(mut p:Loaded,budget:usize)->Result<Probe,String>{
    if !(1..=MAX_STEPS).contains(&budget){return Err("DOS step budget must be 1..1000000".into());}
    let(mut completed,mut pil_steps,mut dos_calls)=(0,0,0);let mut execution_steps=0;let mut output=Vec::new();let mut instructions=BTreeMap::new();
    let mut cache=super::cache::DecodeCache::default();
    let stop=loop{
        if execution_steps==budget{break Stop::Budget;}
        if let Ok(encoded)=bytes_at(&p,2){if encoded[0]==0xcd{
            let vector=encoded[1];let key=format!("DOS INT{vector:02x}/AH{:02x}",(p.cpu.regs[0]>>8)&255);
            match service(&mut p,vector,&mut output){
                Ok(exit)=>{p.cpu.eip=(p.cpu.eip+2)&0xffff;p.image.retired();completed+=1;execution_steps+=1;dos_calls+=1;*instructions.entry(key).or_insert(0)+=1;
                    if let Some(code)=exit{break Stop::Exit(code);}continue;},
                Err(message)=>break Stop::Service(message),
            }
        }}
        let decoded=match cache.fetch(&p.image,&p.ram,p.cpu.eip){
            Ok(d)=>d,Err(GuestError::Execution(e))=>break Stop::Fault(format!("{e:?}")),
            Err(GuestError::Decode(detail))=>break if detail.starts_with("fetch outside loaded image"){Stop::Boundary(detail)}else{Stop::MissingEncoding{key:opcode_key(&site_bytes(&p)),detail}},
        };
        let old_ip=p.cpu.eip;let old_count=p.cpu.regs[1];
        let result=match &decoded.operation{
            Operation::Pil{program,dst,src,flow,address}=>{
                let mut machine=p.image.segment_state();let result=super::execute_machine(program,&mut p.cpu,*dst,*src,*flow,&mut p.ram,*address,&mut machine);
                p.image.set_segment_state(machine);p.selectors=machine.selectors;result
            },
            Operation::Halt=>break Stop::Monitor("HLT reached;monitor stop is not DOS process termination".into()),
            _=>break Stop::Monitor("monitor-port I/O is outside the DOS probe shim".into()),
        };
        if let Err(e)=result{break Stop::Fault(format!("{e:?}"));}
        execution_steps+=1;pil_steps+=1;
        let repeated=matches!(&decoded.operation,Operation::Pil{program,..}if program.partial);
        let partial=repeated&&p.cpu.eip==old_ip;
        if repeated&&p.cpu.regs[1]!=old_count{*instructions.entry(format!("{} elements",decoded.label)).or_insert(0)+=1;}
        if !partial{completed+=1;let key=if repeated{decoded.label.clone()}else{decoded.label.split_whitespace().next().unwrap_or("?").to_string()};*instructions.entry(key).or_insert(0)+=1;}

    };
    let stop_bytes=site_bytes(&p);let linear=p.image.segments[SegmentRegister::Cs.index()].base.checked_add(p.cpu.eip);
    let file_offset=linear.and_then(|a|a.checked_sub(p.module_base)).filter(|n|(*n as usize)<p.module_bytes).map(|n|n as usize+p.header_bytes);
    Ok(Probe{loaded:p,stop,completed,pil_steps,dos_calls,output,instructions,stop_bytes,file_offset})
}
fn ranked(counts:&BTreeMap<String,usize>)->Vec<(&String,&usize)>{let mut v:Vec<_>=counts.iter().collect();v.sort_by(|a,b|b.1.cmp(a.1).then(a.0.cmp(b.0)));v}

    if !(1..=MAX_STEPS).contains(&budget){return Err("DOS step budget must be 1..1000000".into());}
    let mut report=format!("DOS executable probe v{}; step budget={} per program: concrete PIL interpreter plus minimal host DOS-call shim.\nSHIM PROFILE version6.22, OEM FF, serial0, no ROM; PSP:0040 supplies AH30 version. Not a running DOS kernel.\n",env!("CARGO_PKG_VERSION"),budget);
    let mut inputs:Vec<(String,Result<Vec<u8>,String>)>=Vec::new();
    let mut fingerprints:BTreeMap<u64,Vec<Vec<u8>>>=BTreeMap::new();
    for path in paths{match super::containers::open(Path::new(path)){
        Ok(Some(archive))=>{
            let mut candidates=0;let mut added=0;

        writeln!(report,"OUTPUT (escaped bytes): {}",result.output.iter().map(|b|std::ascii::escape_default(*b).map(char::from).collect::<String>()).collect::<String>()).unwrap();
    }

}

//! Bounded single-process DOS MCB arena. No UMBs, EXEC, TSR or device ownership.
use super::{Loaded,Ram,MemWidth,PSP,LIMIT};
const END:u16=(LIMIT/16)as u16;
const MAX_BLOCKS:usize=4096;
#[derive(Clone,Debug,PartialEq,Eq)]
struct Block{header:u16,size:u16,owner:u16}
#[derive(Clone,Debug,PartialEq,Eq)]
pub(super) struct Arena{blocks:Vec<Block>}
impl Arena{
    pub(super) fn new(program_end:u16)->Self{
        let mut blocks=vec![Block{header:PSP-1,size:program_end-PSP,owner:PSP}];
        if program_end<END{blocks.push(Block{header:program_end,size:END-program_end-1,owner:0});}
        Self{blocks}
    }
    fn headers(&self)->Vec<(u32,[u8;5])>{
        self.blocks.iter().enumerate().map(|(i,b)|{let mut bytes=[0;5];bytes[0]=if i+1==self.blocks.len(){b'Z'}else{b'M'};bytes[1..3].copy_from_slice(&b.owner.to_le_bytes());bytes[3..5].copy_from_slice(&b.size.to_le_bytes());(u32::from(b.header)*16,bytes)}).collect()
    }
    pub(super) fn write_initial(&self,ram:&mut Ram)->Result<(),String>{
        for(at,bytes)in self.headers(){ram.write_bytes(at,&bytes).map_err(|e|format!("DOS initial MCB: {e:?}"))?;}Ok(())
    }
    fn valid(&self,ram:&Ram)->bool{
        self.headers().iter().all(|(at,bytes)|bytes.iter().enumerate().all(|(n,&b)|ram.load(*at+n as u32,MemWidth::Byte)==Ok(u32::from(b))))
    }
    fn coalesce(&mut self){
        let mut i=0;while i+1<self.blocks.len(){if self.blocks[i].owner==0&&self.blocks[i+1].owner==0{let b=self.blocks.remove(i+1);self.blocks[i].size+=b.size+1;}else{i+=1;}}
    }
    fn largest(&self)->u16{self.blocks.iter().filter(|b|b.owner==0).map(|b|b.size).max().unwrap_or(0)}
    fn owned(&self,segment:u16)->Option<usize>{self.blocks.iter().position(|b|b.header+1==segment&&b.owner==PSP)}
}
fn low(register:&mut u32,value:u16){*register=(*register&0xffff0000)|u32::from(value);}
fn result(p:&mut Loaded,error:Option<u16>,bx:Option<u16>,allocated:Option<u16>){
    p.cpu.eflags=(p.cpu.eflags&!1)|u32::from(error.is_some());p.cpu.undefined_flags&=!1;
    if let Some(ax)=error.or(allocated){low(&mut p.cpu.regs[0],ax);}
    if let Some(bx)=bx{low(&mut p.cpu.regs[3],bx);}
}
pub(super) fn memory_service(p:&mut Loaded,function:u8)->Result<(),String>{
    // MCB fields are mirrored in RAM. Unsupported direct chain edits return DOS7;
    // do not silently overwrite a corrupted header from trusted host metadata.
    if !p.arena.valid(&p.ram){result(p,Some(7),None,None);return Ok(());}
    let mut next=p.arena.clone();let requested=p.cpu.regs[3]as u16;
    let segment=p.image.selectors[0];let(mut error,mut bx,mut allocated)=(None,None,None);
    match function{
        0x48=>{
            next.coalesce();
            if let Some(i)=next.blocks.iter().position(|b|b.owner==0&&b.size>=requested){
                let available=next.blocks[i].size;let header=next.blocks[i].header;
                next.blocks[i].owner=PSP;next.blocks[i].size=requested;allocated=Some(header+1);
                if requested<available{next.blocks.insert(i+1,Block{header:header+1+requested,size:available-requested-1,owner:0});}
            }else{error=Some(8);bx=Some(next.largest());}
        }
        0x49=>{
            if let Some(i)=next.owned(segment){next.blocks[i].owner=0;}else{error=Some(9);}
            // DOS2..6 defers coalescing until allocation/resize.
        }
        0x4a=>{
            next.coalesce();
            if let Some(i)=next.owned(segment){
                let mut available=next.blocks[i].size;
                if i+1<next.blocks.len()&&next.blocks[i+1].owner==0{let free=next.blocks.remove(i+1);available+=free.size+1;}
                allocated=Some(segment);
                let actual=requested.min(available);let header=next.blocks[i].header;next.blocks[i].size=actual;
                if actual<available{next.blocks.insert(i+1,Block{header:header+1+actual,size:available-actual-1,owner:0});}
                // DOS2..6 expands to the maximum even on insufficient-memory error.
                if requested>available{error=Some(8);bx=Some(available);}
            }else{error=Some(9);}
        }
        _=>return Err("invalid DOS memory service dispatch".into()),
    }
    if next.blocks.len()>MAX_BLOCKS{return Err("DOS probe MCB block-count limit reached".into());}
    let headers:Vec<_>=next.headers().into_iter().filter(|(at,bytes)|bytes.iter().enumerate().any(|(n,&b)|p.ram.load(*at+n as u32,MemWidth::Byte)!=Ok(u32::from(b)))).collect();
    // PSP:0002 retains its load-time value, as in the FreeDOS handler.
    // Report current ownership separately from that original PSP field.
    let program_end=next.owned(PSP).map(|i|(u32::from(PSP)+u32::from(next.blocks[i].size))*16).unwrap_or(u32::from(PSP)*16);
    for(at,bytes)in &headers{for n in 0..bytes.len(){p.ram.check_write(*at+n as u32,MemWidth::Byte).map_err(|e|format!("DOS MCB write: {e:?}"))?;}}
    for(at,bytes)in headers{p.ram.write_bytes(at,&bytes).map_err(|e|format!("DOS MCB commit: {e:?}"))?;}
    p.arena=next;p.allocated_end=program_end;result(p,error,bx,allocated);Ok(())
}

//! Context supplied by a future decoder; no instruction fetch is performed.
#[derive(Clone, Copy, Debug)]
pub struct FlowContext {
    pub next_eip: u32,
    pub displacement: i32,
    pub displacement_bits: u8,
    pub code_limit: u32,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ExecError {
    MemoryAccess { eip:u32, address:u32, bytes:u8, write:bool },
    StackSegment { eip: u32, error_code: u16 },
    InvalidBinding,
    InvalidContext,
    GeneralProtection { eip: u32, error_code: u16 },
}

impl FlowContext {
    pub fn validate(self, operand_width: u8, uses_displacement: bool) -> Result<(), ExecError> {
        if uses_displacement {
            let fits = match self.displacement_bits {
                8 => (-128..=127).contains(&self.displacement),
                16 => (-32768..=32767).contains(&self.displacement),
                32 => true,
                _ => false,
            };
            if !fits || (self.displacement_bits != 8 && self.displacement_bits != operand_width) {
                return Err(ExecError::InvalidContext);
            }
        }
        Ok(())
    }
}

const CONDITIONS:[&str;16]=["o","no","b","ae","e","ne","be","a","s","ns","p","np","l","ge","le","g"];
#[derive(Clone,Debug,PartialEq,Eq)]
pub enum GuestError { Decode(String), Execution(ExecError) }

pub enum Operation {

#[derive(Clone,Debug,PartialEq,Eq)]
pub struct Image {pub fetch_base:u32,pub start:u32,pub end:u32,pub segments:[SegmentCache;6],pub selectors:[u16;6],pub msw:u16,pub real_mode:bool,pub ss_shadow:u8,pub sti_shadow:u8,pub flag_checkpoint:Option<(u32,u32,u32)>}
impl Image {
    // Keep the imported physical code window fixed when CS changes.
    pub fn contains_offset(&self,offset:u32)->bool{
        let at=u64::from(self.segments[1].base)+u64::from(offset);
        at>=u64::from(self.fetch_base)+u64::from(self.start)&&at<u64::from(self.fetch_base)+u64::from(self.end)
    }
    pub fn retired(&mut self){let mut s=self.segment_state();s.retired();self.set_segment_state(s);}
    pub fn segment_state(&self)->SegmentState{SegmentState{msw:self.msw,segments:self.segments,selectors:self.selectors,real_mode:self.real_mode,ss_shadow:self.ss_shadow,sti_shadow:self.sti_shadow,flag_checkpoint:self.flag_checkpoint}}
    pub fn set_segment_state(&mut self,s:SegmentState){self.msw=s.msw;self.segments=s.segments;self.selectors=s.selectors;self.real_mode=s.real_mode;self.ss_shadow=s.ss_shadow;self.sti_shadow=s.sti_shadow;self.flag_checkpoint=s.flag_checkpoint;}

    pub fn load(bytes:&[u8])->Result<(Self,Ram),String> {
        if bytes.is_empty() || bytes.len()>RAM_BYTES-ENTRY as usize {return Err("image must contain 1..3840 bytes".into());}
        let mut ram=Ram::new(RAM_BYTES);ram.write_bytes(ENTRY,bytes).map_err(|e|format!("{e:?}"))?;
        let mut segments=[SegmentCache::real_mode(0);6];
        // Ordinary real-mode CS permits data reads and writes too.
        // Protected-mode callers supply Code caches with their own permissions.
        segments[SegmentRegister::Ds.index()]=SegmentCache::real_mode(0x10);
        segments[SegmentRegister::Ss.index()]=SegmentCache::real_mode(0x20);
        segments[SegmentRegister::Fs.index()]=SegmentCache::real_mode(0x30);
        Ok((Self{fetch_base:0,start:ENTRY,end:ENTRY+bytes.len() as u32,segments,selectors:[0,0,0x20,0x10,0x30,0],msw:0x10,real_mode:true,ss_shadow:0,sti_shadow:0,flag_checkpoint:None},ram))
    }
    pub fn fetch_signature(&self,ram:&Ram,eip:u32,length:u32)->Result<[u8;15],GuestError> {
        if length==0 || length>15{return Err(GuestError::Decode("invalid cached instruction length".into()));}
        let mut reader=Reader{image:self,ram,start:eip,pos:eip};let mut bytes=[0u8;15];
        for byte in &mut bytes[..length as usize] {*byte=reader.byte()?;}
        Ok(bytes)
    }
    pub fn decode(&self,ram:&Ram,eip:u32)->Result<Decoded,GuestError> {
        let mut r=Reader{image:self,ram,start:eip,pos:eip};
        let mut wide=false;let mut addr32=false;let mut override_segment=None;let mut repeat=None;let mut locked=false;
        let mut opcode=loop {
            let b=r.byte()?;
            match b {
                0xf2|0xf3 if repeat.is_none()=>repeat=Some(b),
                0xf0 if !locked=>locked=true,
                0x66 if !wide=>wide=true,0x67 if !addr32=>addr32=true,
                0x26|0x2e|0x36|0x3e|0x64|0x65 if override_segment.is_none()=>override_segment=Some(match b{0x26=>SegmentRegister::Es,0x2e=>SegmentRegister::Cs,0x36=>SegmentRegister::Ss,0x3e=>SegmentRegister::Ds,0x64=>SegmentRegister::Fs,_=>SegmentRegister::Gs}),
                _=>break b,
            }
        };
        if let Some(prefix)=repeat{if !matches!(opcode,0xa4|0xa5|0xaa|0xab|0xac|0xad|0xa6|0xa7|0xae|0xaf)||(prefix==0xf2&&!matches!(opcode,0xa6|0xa7|0xae|0xaf)){return Err(GuestError::Decode("unsupported repeat-prefix/instruction combination".into()));}}
        if locked&&!matches!(opcode,0x86|0x87){return Err(GuestError::Decode("LOCK requires memory XCHG in this profile".into()));}
        // Normalize alternative register encodings while retaining consumed bytes.
        let mut push_selector=None;let mut pop_selector=None;let mut call_indirect=None;let mut memory_pop=None;let mut memory_push=None;let mut far_indirect=None;
        if matches!(opcode,0xff|0x8f){
            let m=r.byte()?;let extension=(m>>3)&7;
            if opcode==0xff&&extension==2{call_indirect=Some(m);opcode=0xe8;}
            else if opcode==0xff&&extension==3{if m>>6==3{return Err(GuestError::Decode("far CALL requires a memory pointer".into()));}far_indirect=Some(m);opcode=0x9a;}
            else if opcode==0x8f || extension==6{
                if opcode==0x8f && extension!=0{return Err(GuestError::Decode("unsupported POP group extension".into()));}
                if m>>6!=3{
                    if opcode==0x8f{memory_pop=Some(m);}else{memory_push=Some(m);}
                }else{opcode=(if opcode==0x8f{0x58}else{0x50})|(m&7);}
            }else{r.pos-=1;}
        }
        if opcode==0x0f{
            let second=r.byte()?;
            if matches!(second,0xa1|0xa9){pop_selector=Some(if second==0xa1{SegmentRegister::Fs}else{SegmentRegister::Gs});opcode=0x07;}
            else if matches!(second,0xa0|0xa8){push_selector=Some(if second==0xa0{SegmentRegister::Fs}else{SegmentRegister::Gs});opcode=0x06;}
            else{r.pos-=1;}
        }
        let mut w=if wide {32}else{16};let mut dst=view(0,w);let mut src=dst;let mut address=None;let mut disp=0;let mut disp_bits=8;
        let (label,program)=match opcode {
            0xff if memory_push.is_some()=>{
                let m=memory_push.unwrap();
                let effective=r.effective(m,addr32)?;address=Some(AddressContext{effective,override_segment,segments:self.segments});
                let stack_bits=if matches!(self.segments[2].access,SegmentAccess::Data{big:true,..}){32}else{16};
                let text=description("push_memory").map_err(GuestError::Decode)?.replace("$S",&stack_bits.to_string()).replace("$BYTES",&(w/8).to_string());
                (format!("PUSH {}",operand_name(dst,address)),compile(&instantiate(&text,w),w))
            }
            0x8f=>{
                let m=memory_pop.ok_or_else(||GuestError::Decode("invalid memory POP".into()))?;
                let effective=r.effective(m,addr32)?;address=Some(AddressContext{effective,override_segment,segments:self.segments});
                let stack_bits=if matches!(self.segments[2].access,SegmentAccess::Data{big:true,..}){32}else{16};
                let esp_based=matches!(effective,EffectiveAddress::Bits32{base:Some(Gpr::Esp),..});
                // Intel documents this wrap case as processor-family-specific.
                // Reject it until a legacy386 hardware vector fixes the policy.
                let guard=if stack_bits==16&&esp_based{"ult wrapped new_sp old_sp\nnot allowed wrapped\nrequire allowed"}else{""};
                let text=include_str!("../isa/i386/pop_memory.pil").replace("$S",&stack_bits.to_string()).replace("$BYTES",&(w/8).to_string()).replace("$HIGH_MASK",if stack_bits==16{"4294901760"}else{"0"}).replace("$WRAP_GUARD",guard);
                (format!("POP {}",operand_name(dst,address)),compile(&instantiate(text,w),w))
            }
            0xc9=>{
                let stack_bits=if matches!(self.segments[2].access,SegmentAccess::Data{big:true,..}){32}else{16};
                let text=description("leave").map_err(GuestError::Decode)?.replace("$S",&stack_bits.to_string()).replace("$BYTES",&(w/8).to_string());
                ("LEAVE".into(),compile(&instantiate(text,w),w))
            }
            0xe3=>{
                disp=r.byte()? as i8 as i32;
                let name=if addr32{"JECXZ"}else{"JCXZ"};
                let text=include_str!("../isa/i386/jcxz.pil").replace("$A",if addr32{"32"}else{"16"});
                (format!("{name} 0x{:08x}",r.pos.wrapping_add(disp as u32)&super::mask(w)as u32),compile(&instantiate(text,w),w))
            }
            0x9c|0x9d=>{
                let stack_bits=if matches!(self.segments[2].access,SegmentAccess::Data{big:true,..}){32}else{16};
                let name=if opcode==0x9c{"pushf"}else{"popf"};
                let text=description(name).map_err(GuestError::Decode)?.replace("$S",&stack_bits.to_string()).replace("$BYTES",&(w/8).to_string());
                (format!("{}{}",name.to_ascii_uppercase(),if w==32{"D"}else{""}),compile(&instantiate(text,w),w))
            }
            0xfa|0xfb=>{
                let name=if opcode==0xfa{"cli"}else{"sti"};
                (name.to_ascii_uppercase(),compile(&description(name).map_err(GuestError::Decode)?,w))
            }
            0x9a=>{
                let stack_bits=if matches!(self.segments[2].access,SegmentAccess::Data{big:true,..}){32}else{16};
                let (target,label)=if let Some(m)=far_indirect{
                    let effective=r.effective(m,addr32)?;address=Some(AddressContext{effective,override_segment,segments:self.segments});
                    let seg=effective.resolve(&[0;8],override_segment)?.segment.index();
                    (format!("read_address operand\nzext operand_wide operand 64\ncheck_segment {seg} operand_wide {}\nload target_value operand {w}\nconst slot 32 {}\nadd selector_at operand slot\nload selector selector_at 16",w/8+2,w/8),operand_name(dst,address))
                }else{
                    let offset=r.unsigned(w)?;let selector=r.unsigned(16)?;
                    (format!("const target_value {w} {offset}\nconst selector 16 {selector}"),format!("{selector:04x}:{offset:08x}"))
                };
                let text=description("call_far").map_err(GuestError::Decode)?.replace("$FAR_TARGET",&target).replace("$S",&stack_bits.to_string()).replace("$BYTES",&(w/8).to_string()).replace("$FRAME",&(w/4).to_string());
                (format!("CALL FAR {label}"),compile(&instantiate(text,w),w))
            }
            0xe8=>{
                let stack_bits=if matches!(self.segments[2].access,SegmentAccess::Data{big:true,..}){32}else{16};
                let (target,label)=if let Some(m)=call_indirect{
                    dst=view((m&7)as usize,w);src=dst;
                    if m>>6!=3{let effective=r.effective(m,addr32)?;address=Some(AddressContext{effective,override_segment,segments:self.segments});(format!("read_address operand\nload target_value operand {w}"),operand_name(dst,address))}
                    else{(format!("read target_value dst {w}"),register_name(dst).into())}
                }else{
                    disp_bits=w;let raw=r.unsigned(w)?;disp=if w==16{raw as i16 as i32}else{raw as i32};
                    (format!("read_disp delta {w}\nadd target_value return_offset delta"),format!("0x{:08x}",r.pos.wrapping_add(disp as u32)&super::mask(w)as u32))
                };
                let text=include_str!("../isa/i386/call.pil").replace("$TARGET",&target).replace("$S",&stack_bits.to_string()).replace("$BYTES",&(w/8).to_string());
                (format!("CALL {label}"),compile(&instantiate(text,w),w))
            }
            0xca|0xcb=>{
                let cleanup=if opcode==0xca{r.unsigned(16)?}else{0};let stack_bits=if matches!(self.segments[2].access,SegmentAccess::Data{big:true,..}){32}else{16};
                let text=include_str!("../isa/i386/retf.pil").replace("$CLEANUP",&cleanup.to_string()).replace("$S",&stack_bits.to_string()).replace("$BYTES",&(w/8).to_string()).replace("$FRAME",&(w/4).to_string());
                (if opcode==0xca{format!("RETF 0x{cleanup:04x}")}else{"RETF".into()},compile(&instantiate(text,w),w))
            }
            0xc2|0xc3=>{
                let cleanup=if opcode==0xc2{r.unsigned(16)?}else{0};let stack_bits=if matches!(self.segments[2].access,SegmentAccess::Data{big:true,..}){32}else{16};
                let text=include_str!("../isa/i386/ret.pil").replace("$CLEANUP",&cleanup.to_string()).replace("$S",&stack_bits.to_string()).replace("$BYTES",&(w/8).to_string());
                (if opcode==0xc2{format!("RET 0x{cleanup:04x}")}else{"RET".into()},compile(&instantiate(text,w),w))
            }
            0xa4|0xa5|0xaa|0xab|0xac|0xad=>{
                if opcode&1==0{w=8;}dst=view(0,w);src=dst;let a=if addr32{32}else{16};
                let name=match opcode{0xa4|0xa5=>"movs",0xaa|0xab=>"stos",_=>"lods"};
                let body=match name{"movs"=>include_str!("../isa/i386/movs.pil"),"stos"=>include_str!("../isa/i386/stos.pil"),_=>include_str!("../isa/i386/lods.pil")};
                let body=body.replace("$SEG",&override_segment.unwrap_or(SegmentRegister::Ds).index().to_string()).replace("$A",&a.to_string()).replace("$BYTES",&(w/8).to_string());
                let text=if repeat.is_some(){super::repeat_description(&body,a)}else{body};
                let mnemonic=format!("{}{}",name.to_ascii_uppercase(),match w{8=>"B",16=>"W",_=>"D"});
                (if repeat.is_some(){format!("REP {mnemonic}")}else{mnemonic},compile(&instantiate(text,w),w))
            }
            0xa6|0xa7|0xae|0xaf=>{
                if opcode&1==0{w=8;}dst=view(0,w);src=dst;let a=if addr32{32}else{16};let name=if opcode<0xae{"cmps"}else{"scas"};
                let body=description(name).map_err(GuestError::Decode)?.replace("$SEG",&override_segment.unwrap_or(SegmentRegister::Ds).index().to_string()).replace("$A",&a.to_string()).replace("$BYTES",&(w/8).to_string());
                let text=if let Some(prefix)=repeat{super::compare_repeat_description(&body,a,prefix==0xf3)}else{body};
                let mnemonic=format!("{}{}",name.to_ascii_uppercase(),match w{8=>"B",16=>"W",_=>"D"});
                (if let Some(prefix)=repeat{format!("{} {mnemonic}",if prefix==0xf3{"REPE"}else{"REPNE"})}else{mnemonic},compile(&instantiate(text,w),w))
            }
            0xa0..=0xa3=>{
                if opcode&1==0{w=8;}dst=view(0,w);src=dst;
                let offset=r.unsigned(if addr32{32}else{16})?;
                let effective=if addr32{EffectiveAddress::Bits32{base:None,index:None,scale:1,displacement:offset}}else{EffectiveAddress::Bits16{form:Address16::Direct,displacement:offset as u16}};
                address=Some(AddressContext{effective,override_segment,segments:self.segments});
                let load=opcode&2==0;let text=instantiate(description(if load{"load_ram"}else{"store_ram"}).map_err(GuestError::Decode)?,w);
                let operand=operand_name(dst,address);
                (if load{format!("MOV {},{operand}",register_name(dst))}else{format!("MOV {operand},{}",register_name(src))},compile(&text,w))
            }
            0xc6|0xc7=>{
                if opcode==0xc6{w=8;}
                let m=r.byte()?;if (m>>3)&7!=0{return Err(GuestError::Decode("MOV immediate requires group extension0".into()));}
                dst=view((m&7)as usize,w);src=dst;
                let memory=m>>6!=3;
                if memory{let effective=r.effective(m,addr32)?;address=Some(AddressContext{effective,override_segment,segments:self.segments});}
                let immediate=r.unsigned(w)?;
                let text=if memory{format!("read_address addr\nconst v {w} {immediate}\nstore addr v\npreserve_flags all\nretire")}else{format!("const v {w} {immediate}\nwrite dst v\npreserve_flags all\nretire")};
                (format!("MOV {},0x{immediate:0digits$x}",operand_name(dst,address),digits=(w/4)as usize),compile(&text,w))
            }
            0xc4=>{
                if !self.real_mode{return Err(GuestError::Decode("LES requires real-mode context".into()));}
                let m=r.byte()?;if m>>6==3{return Err(GuestError::Decode("LES requires a memory far pointer".into()));}
                dst=view(((m>>3)&7)as usize,w);src=dst;
                let effective=r.effective(m,addr32)?;
                let segment=effective.resolve(&[0;8],override_segment)?.segment.index();
                address=Some(AddressContext{effective,override_segment,segments:self.segments});
                let text=include_str!("../isa/i386/les.pil").replace("$SEG",&segment.to_string()).replace("$SPAN",&(w/8+2).to_string()).replace("$BYTES",&(w/8).to_string());
                (format!("LES {},{}",register_name(dst),operand_name(src,address)),compile(&instantiate(text,w),w))
            }
            0x8c|0x8e=>{
                let m=r.byte()?;let segment=((m>>3)&7)as usize;
                if segment>=6 || (opcode==0x8e&&segment==1){return Err(GuestError::Decode("invalid segment MOV encoding".into()));}
                if opcode==0x8e&&!self.real_mode{return Err(GuestError::Decode("segment loads require real-mode context".into()));}
                if opcode==0x8c&&wide&&m>>6==3{return Err(GuestError::Decode("386 MOV r32,Sreg upper-half policy is not admitted".into()));}
                w=16;dst=view((m&7)as usize,w);src=dst;
                let seg=["ES","CS","SS","DS","FS","GS"][segment];
                let mut text=if opcode==0x8c{format!("read_segment selector {segment}\nwrite dst selector\npreserve_flags all\nretire")}
                    else{format!("read selector src 16\nwrite_segment {segment} selector\npreserve_flags all\nretire")};
                if m>>6!=3{let effective=r.effective(m,addr32)?;address=Some(AddressContext{effective,override_segment,segments:self.segments});
                    text=if opcode==0x8c{text.replace("write dst selector","read_address addr\nstore addr selector")}else{text.replace("read selector src 16","read_address addr\nload selector addr 16")};}
                let operand=operand_name(dst,address);
                (if opcode==0x8c{format!("MOV {operand},{seg}")}else{format!("MOV {seg},{operand}")},compile(&text,w))
            }
            0x07|0x17|0x1f=>{
                if !self.real_mode{return Err(GuestError::Decode("segment loads require real-mode context".into()));}
                let segment=pop_selector.unwrap_or(match opcode{0x07=>SegmentRegister::Es,0x17=>SegmentRegister::Ss,_=>SegmentRegister::Ds});
                let stack_bits=if matches!(self.segments[2].access,SegmentAccess::Data{big:true,..}){32}else{16};
                address=Some(AddressContext{effective:EffectiveAddress::Bits16{form:Address16::Direct,displacement:0},override_segment:Some(SegmentRegister::Ss),segments:self.segments});
                let text=include_str!("../isa/i386/pop_segment.pil").replace("$SEG",&segment.index().to_string()).replace("$S",&stack_bits.to_string()).replace("$BYTES",&(w/8).to_string());
                (format!("POP {segment:?}").to_ascii_uppercase(),compile(&instantiate(text,w),w))
            }
            0x50..=0x5f|0x68|0x6a|0x06|0x0e|0x16|0x1e=>{
                let stack_bits=match self.segments[SegmentRegister::Ss.index()].access {
                    SegmentAccess::Data{big:true,..}=>32,_=>16,
                };
                // Stack addresses always use SS, independent of 67 or overrides.
                address=Some(AddressContext{effective:EffectiveAddress::Bits16{form:Address16::Direct,displacement:0},override_segment:Some(SegmentRegister::Ss),segments:self.segments});
                let pop=(0x58..=0x5f).contains(&opcode);let reg=(opcode&7)as usize;
                let name=if pop{"pop"}else{"push"};
                let mut text=description(name).map_err(GuestError::Decode)?
                    .replace("$S",&stack_bits.to_string()).replace("$BYTES",&(w/8).to_string()).replace("$REG",&reg.to_string());
                let operand=if (0x50..=0x5f).contains(&opcode){dst=view(reg,w);register_name(dst).to_string()}
                else {
                    let (v,label)=if opcode==0x68{let v=r.unsigned(w)?;(v,format!("0x{v:x}"))}
                    else if opcode==0x6a{let v=r.byte()? as i8 as i32 as u32 & super::mask(w)as u32;(v,format!("0x{v:x}"))}
                    else {let seg=push_selector.unwrap_or(match opcode{0x06=>SegmentRegister::Es,0x0e=>SegmentRegister::Cs,0x16=>SegmentRegister::Ss,_=>SegmentRegister::Ds});(u32::from(self.selectors[seg.index()]),format!("{seg:?}").to_ascii_uppercase())};
                    if matches!(opcode,0x06|0x0e|0x16|0x1e){let seg=push_selector.unwrap_or(match opcode{0x06=>SegmentRegister::Es,0x0e=>SegmentRegister::Cs,0x16=>SegmentRegister::Ss,_=>SegmentRegister::Ds});text=text.replace("read data dst $W",&format!("read_segment selector {}\nzext data selector $W",seg.index()));}
                    else{text=text.replace("read data dst $W",&format!("const data $W {v}"));}label
                };
                (format!("{} {operand}",name.to_ascii_uppercase()),compile(&instantiate(text,w),w))
            }
            0xb0..=0xbf=>{
                if opcode<0xb8 {w=8;}
                dst=view((opcode&7) as usize,w);src=dst;
                let value=r.unsigned(w)?;
                (format!("MOV {},0x{value:0digits$x}",register_name(dst),digits=(w/4) as usize),compile(&format!("const v {w} {value}\nwrite dst v\npreserve_flags all\nretire"),w))
            }
            0x70..=0x7f=>{
                disp=r.byte()? as i8 as i32;
                let name=format!("j{}",CONDITIONS[(opcode&15) as usize]);
                (format!("{} 0x{:08x}",name.to_ascii_uppercase(),r.pos.wrapping_add(disp as u32)&super::mask(w) as u32),compile(&instantiate(description(&name).map_err(GuestError::Decode)?,w),w))
            }
            0xeb|0xe9=>{
                disp_bits=if opcode==0xeb{8}else{w};
                let raw=r.unsigned(disp_bits)?;disp=match disp_bits{8=>raw as i8 as i32,16=>raw as i16 as i32,_=>raw as i32};
                (format!("JMP 0x{:08x}",r.pos.wrapping_add(disp as u32)&super::mask(w) as u32),compile(&instantiate(description("jmp").map_err(GuestError::Decode)?,w),w))
            }
            0x0f=>{
                let second=r.byte()?;

struct Huff{max:usize,table:Vec<(u16,u8)>}
impl Huff{
    fn new(lengths:&[u8])->Result<Self,String>{

        let mut next=[0u32;16];let mut code=0;

        }Ok(Self{max,table})
    }
    fn read(&self,b:&mut Bits)->Result<usize,String>{
        if self.max==0{return Err("empty DEFLATE distance tree used".into());}
        let(symbol,n)=self.table[b.peek(self.max)as usize];if n==0{return Err("invalid DEFLATE code".into());}b.read(n as usize)?;Ok(symbol as usize)
    }

        match kind{

                if out.len()+n>expected{return Err("DEFLATE exceeds declared output size".into());}for _ in 0..n{out.push(b.read(8)?as u8);}},
            1|2=>{let(ll,dd)=trees(&mut b,kind)?;
                loop{match ll.read(&mut b)?{
                    n@0..=255=>{if out.len()==expected{return Err("DEFLATE exceeds declared output size".into());}out.push(n as u8);},
                    256=>break,
                    n@257..=285=>{let n=n-257;let count=LB[n]+b.read(LE[n])?as usize;let d=dd.read(&mut b)?;
                        if d>=30{return Err("reserved DEFLATE distance".into());}let distance=DB[d]+b.read(DE[d])?as usize;
                        if distance>out.len()||out.len()+count>expected{return Err("invalid DEFLATE back-reference/output size".into());}
                        for _ in 0..count{out.push(out[out.len()-distance]);}},
                    _=>return Err("reserved DEFLATE length".into()),
                }}
            },_=>return Err("reserved DEFLATE block type".into()),
        }
        if last!=0{break;}
    }
    if out.len()!=expected||(b.at+7)/8!=data.len(){return Err("DEFLATE size/trailing-byte mismatch".into());}Ok(out)
}

#[derive(Clone, Copy, Debug)]
struct Value { bits: u64, width: u8 }
fn mask(w: u8) -> u64 { if w == 64 { u64::MAX } else { (1u64 << w) - 1 } }
fn value(bits: u64, width: u8) -> Value { Value { bits: bits & mask(width), width } }

#[derive(Clone, Copy, Debug)]
enum Binding { Dst, Src }
#[derive(Clone, Copy, Debug)]

#[derive(Clone, Copy, Debug)]

#[derive(Clone, Copy, Debug)]
enum Flag { CF, PF, AF, ZF, SF, TF, IF, DF, OF, IOPL0, IOPL1, NT }
impl Flag {
    fn bit(self) -> u8 { match self {
        Self::CF => 0, Self::PF => 2, Self::AF => 4,
        Self::ZF => 6, Self::SF => 7, Self::TF => 8, Self::IF => 9, Self::DF => 10, Self::OF => 11, Self::IOPL0 => 12, Self::IOPL1 => 13, Self::NT => 14,
    }}
}
fn parse_flag(name: &str) -> R<Flag> {
    match name {
        "CF" => Ok(Flag::CF), "PF" => Ok(Flag::PF), "AF" => Ok(Flag::AF),
        "TF" => Ok(Flag::TF), "IF" => Ok(Flag::IF), "IOPL0" => Ok(Flag::IOPL0), "IOPL1" => Ok(Flag::IOPL1), "NT" => Ok(Flag::NT),
        "ZF" => Ok(Flag::ZF), "SF" => Ok(Flag::SF), "DF" => Ok(Flag::DF), "OF" => Ok(Flag::OF),
        _ => Err(format!("unknown flag: {name}")),
    }
}

#[derive(Clone, Debug)]
struct Program { tokens: Vec<Token>, widths: Vec<u8>, operand_width: u8, source_width: u8, uses_flow: bool, uses_displacement: bool, uses_memory: bool, partial: bool, checkpoint_flags:bool }

fn number(s: &str) -> R<u64> {
    if let Some(hex) = s.strip_prefix("0x") {
        u64::from_str_radix(hex, 16).map_err(|_| format!("bad number: {s}"))
    } else { s.parse().map_err(|_| format!("bad number: {s}")) }
}
fn width(s: &str) -> R<u8> {
    let w = number(s)?;
    if (1..=64).contains(&w) { Ok(w as u8) } else { Err("width must be 1..64".into()) }
}
fn slot(names: &HashMap<String, usize>, s: &str) -> R<usize> {
    names.get(s).copied().ok_or_else(|| format!("undefined value: {s}"))
}
fn define(names: &mut HashMap<String, usize>, widths: &mut Vec<u8>, s: &str, w: u8) -> R<usize> {
    if names.contains_key(s) { return Err(format!("duplicate value: {s}")); }
    let id = widths.len();
    widths.push(w); names.insert(s.to_owned(), id); Ok(id)
}
fn compile(source: &str, operand_width: u8) -> R<Program> {
    if ![8,16,32].contains(&operand_width) { return Err("operand width must be 8,16,32".into()); }
    let mut names = HashMap::new();
    let mut widths = Vec::new();
    let mut tokens = Vec::new();
    let mut retired = false;
    let mut wrote = false;
    let mut wrote_src = false;
    let mut branched = false;
    let mut stored = false;
    let mut store_count=0;
    let mut segment_written=false;
    let mut shadow_written=false;
    let mut flags_seen = 0u32;
    let mut source_width=None;
    for (line_no, raw) in source.lines().enumerate() {
        let line = raw.split('#').next().unwrap_or("").trim();
        if line.is_empty() { continue; }
        let fields: Vec<_> = line.split_whitespace().collect();
        let parsed: R<Token> = (|| {
            if retired { return Err("tokens after retire".into()); }
            let t = match fields.as_slice() {
                ["require_width", required] => {
                    let available=if *required=="16_or_32" { [16,32].contains(&operand_width) }
                        else {width(required)?==operand_width};
                    if !available { return Err("instruction width not available".into()); }
                    Token::PreserveFlags(0)
                }
                ["preserve_flag", name] => {
                    let f=parse_flag(name)?; let m=1u32<<f.bit();
                    if flags_seen&m!=0 {return Err("duplicate flag classification".into());}
                    flags_seen|=m; Token::PreserveFlags(m)
                }
                ["preserve_flags", what] => {
                    let m=match *what {
                        "all" => ARITH_FLAGS,
                        "remaining" => ARITH_FLAGS & !flags_seen,
                        _ => return Err("expected all or remaining".into()),
                    };
                    if flags_seen&m!=0 {return Err("duplicate flag classification".into());}
                    flags_seen|=m; Token::PreserveFlags(m)
                }
                ["undef_flag", name] => {
                    let f=parse_flag(name)?; let m=1u32<<f.bit();
                    if flags_seen&m!=0 {return Err("duplicate flag classification".into());}
                    flags_seen|=m; Token::UndefFlag(f)
                }
                ["flag_if", name, input, defined, active] => {
                    let f=parse_flag(name)?;let m=1u32<<f.bit();
                    if flags_seen&m!=0{return Err("duplicate flag classification".into());}
                    let a=slot(&names,input)?;let d=slot(&names,defined)?;let c=slot(&names,active)?;
                    if [a,d,c].iter().any(|&n|widths[n]!=1){return Err("flag_if requires three bv<1> inputs".into());}
                    flags_seen|=m;Token::FlagIf(f,a,d,c)
                }
                [op @ ("shl"|"lshr"|"ashr"),out,input,count]=>{
                    let a=slot(&names,input)?;let b=slot(&names,count)?;let w=widths[a];
                    let kind=match *op{"shl"=>Shift::Left,"lshr"=>Shift::Right,_=>Shift::Arithmetic};
                    Token::Shift(define(&mut names,&mut widths,out,w)?,kind,a,b)
                }
                ["ult",out,left,right]=>{
                    let a=slot(&names,left)?;let b=slot(&names,right)?;
                    if widths[a]!=widths[b]{return Err("ult width mismatch".into());}
                    Token::Less(define(&mut names,&mut widths,out,1)?,a,b)
                }
                ["select", out, predicate, yes, no] => {
                    let c=slot(&names,predicate)?; let a=slot(&names,yes)?; let b=slot(&names,no)?;
                    if widths[c]!=1 {return Err("select predicate needs bv<1>".into());}
                    if widths[a]!=widths[b] {return Err("select arm width mismatch".into());}
                    let w=widths[a];
                    Token::Select(define(&mut names,&mut widths,out,w)?,c,a,b)
                }
                ["write", "src", input] | ["write_if", "src", input, _] => {
                    let a=slot(&names,input)?;let w=widths[a];
                    if ![8,16,32].contains(&w){return Err("source write width must be 8,16,32".into());}
                    if source_width.is_some_and(|old|old!=w){return Err("inconsistent source binding widths".into());}
                    source_width=Some(w);
                    if wrote_src{return Err("multiple source writes".into());}wrote_src=true;
                    if fields[0]=="write_if"{let c=slot(&names,fields[3])?;
                        if widths[c]!=1{return Err("write_if predicate needs bv<1>".into());}
                        Token::WriteSourceIf(a,c)
                    }else{Token::WriteSource(a)}
                }
                ["write_if", "dst", input, predicate] => {
                    let a=slot(&names,input)?; let c=slot(&names,predicate)?;
                    if widths[a]!=operand_width {return Err("write width mismatch".into());}
                    if widths[c]!=1 {return Err("write_if predicate needs bv<1>".into());}
                    if wrote {return Err("multiple destination writes".into());}
                    wrote=true; Token::WriteIf(a,c)
                }
                ["checkpoint_flags"]=>{
                    if tokens.iter().any(|t|!matches!(t,Token::PreserveFlags(0))){return Err("checkpoint_flags must precede all operations".into());}
                    Token::CheckpointFlags
                }
                ["read_flags", out] => Token::ReadFlags(define(&mut names,&mut widths,out,32)?),
                ["require_real_mode"] => Token::RequireRealMode,
                ["interrupt_shadow_if", input] => {
                    let c=slot(&names,input)?;
                    if widths[c]!=1||shadow_written{return Err("one bv1 interrupt shadow effect required".into());}
                    shadow_written=true;Token::InterruptShadowIf(c)
                }
                ["read_segment", out, register] => {
                    let reg=number(register)?;if reg>=6{return Err("segment index must be 0..5".into());}
                    Token::ReadSegment(define(&mut names,&mut widths,out,16)?,reg as usize)
                }
                ["write_segment", register, input] => {
                    let reg=number(register)?;let a=slot(&names,input)?;
                    if reg>=6||reg==1||widths[a]!=16||segment_written{return Err("one non-CS segment write of bv16 is required".into());}
                    segment_written=true;Token::WriteSegment(reg as usize,a)
                }
                ["read_gpr", out, register, bits] => {
                    let reg=number(register)? as usize;let w=width(bits)?;
                    if reg>=8 || ![8,16,32].contains(&w){return Err("read_gpr needs register 0..7 and width8/16/32".into());}
                    Token::ReadGpr(define(&mut names,&mut widths,out,w)?,reg,w)
                }
                ["write_gpr", register, input] => {
                    let reg=number(register)? as usize;let a=slot(&names,input)?;let w=widths[a];
                    if reg>=8 || ![8,16,32].contains(&w){return Err("write_gpr needs register 0..7 and bv8/16/32".into());}
                    Token::WriteGpr(reg,a,w)
                }
                ["load_segment", out, register, address, bits]=>{
                    if stored{return Err("loads after a staged store are not supported".into());}
                    let segment=number(register)?;let a=slot(&names,address)?;let w=width(bits)?;
                    if segment>=6||widths[a]!=32{return Err("load_segment needs segment0..5 and bv32 offset".into());}
                    let mem=MemWidth::from_bits(w).ok_or("memory width must be8/16/32")?;
                    Token::LoadSegment(define(&mut names,&mut widths,out,w)?,a,mem,segment as usize)
                }
                ["store_segment", register, address, input]=>{
                    let segment=number(register)?;let a=slot(&names,address)?;let v=slot(&names,input)?;
                    if segment>=6||widths[a]!=32||store_count>=2{return Err("at most two stores with segment0..5 and bv32 offset required".into());}
                    let mem=MemWidth::from_bits(widths[v]).ok_or("memory width must be8/16/32")?;stored=true;store_count+=1;Token::StoreSegment(a,v,mem,segment as usize)
                }
                ["retire_if", predicate]=>{
                    let c=slot(&names,predicate)?;
                    if widths[c]!=1||tokens.iter().any(|t|matches!(t,Token::Write(..)|Token::WriteIf(..)|Token::WriteSource(..)|Token::WriteSourceIf(..)|Token::WriteGpr(..)|Token::WriteSegment(..)|Token::InterruptShadowIf(..)|Token::Store(..)|Token::StoreSegment(..)|Token::Flag(..)|Token::FlagIf(..)|Token::UndefFlag(..)|Token::BranchIf(..)|Token::FarBranch(..)|Token::ResumeIf(..))){return Err("retire_if requires bv1 and no earlier effects".into());}
                    Token::RetireIf(c)
                }
                ["resume_if", predicate]=>{
                    let c=slot(&names,predicate)?;if widths[c]!=1||branched{return Err("one resume/branch effect with bv1 required".into());}branched=true;Token::ResumeIf(c)
                }
                ["read_msw", out]=>Token::ReadMsw(define(&mut names,&mut widths,out,16)?),
                ["require", predicate]=>{
                    let c=slot(&names,predicate)?;if widths[c]!=1{return Err("require needs bv1 predicate".into());}Token::Require(c)
                }
                ["read_address_after", out, register, input]=>{
                    let reg=number(register)?;let a=slot(&names,input)?;
                    if reg>=8||widths[a]!=32{return Err("read_address_after requires register0..7 and bv32 replacement".into());}
                    Token::ReadAddressAfter(define(&mut names,&mut widths,out,32)?,reg as usize,a)
                }
                ["read_address", out] => Token::ReadAddress(define(&mut names,&mut widths,out,32)?),
                ["load", out, address, bits] => {
                    if stored {return Err("loads after a staged store are not supported".into());}
                    let a=slot(&names,address)?;
                    if widths[a]!=32 {return Err("memory address must be bv<32>".into());}
                    let w=width(bits)?;let mem=MemWidth::from_bits(w).ok_or("memory width must be 8,16,32")?;
                    Token::Load(define(&mut names,&mut widths,out,w)?,a,mem)
                }
                ["store", address, input] => {
                    let a=slot(&names,address)?;let v=slot(&names,input)?;
                    if widths[a]!=32 {return Err("memory address must be bv<32>".into());}
                    let mem=MemWidth::from_bits(widths[v]).ok_or("memory width must be 8,16,32")?;
                    if store_count>=2 {return Err("at most two staged RAM stores are supported".into());}
                    stored=true;store_count+=1;Token::Store(a,v,mem)
                }
                ["read_ip", out, which, w] => {
                    let next=match *which {"next"=>true,"current"=>false,_=>return Err("read_ip needs current or next".into())};
                    let w=width(w)?;
                    Token::ReadIp(define(&mut names,&mut widths,out,w)?,next,w)
                }
                ["read_disp", out, w] => {
                    let w=width(w)?;
                    if w!=operand_width || ![16,32].contains(&w) {return Err("read_disp needs the 16/32-bit operand width".into());}
                    Token::ReadDisp(define(&mut names,&mut widths,out,w)?,w)
                }
                ["check_segment", register, offset, bytes]=>{
                    let reg=number(register)?;let a=slot(&names,offset)?;let n=number(bytes)?;
                    if reg>=6||widths[a]!=64||n==0||n>u32::MAX as u64{return Err("check_segment requires segment0..5,bv64 offset,positive u32 byte count".into());}
                    Token::CheckSegment(reg as usize,a,n as u32)
                }
                ["far_branch", selector, offset]=>{
                    let a=slot(&names,selector)?;let b=slot(&names,offset)?;
                    if widths[a]!=16||widths[b]!=32||branched||segment_written||![16,32].contains(&operand_width){return Err("one far_branch requires bv16 selector,bv32 offset and no other segment/branch effect".into());}
                    branched=true;segment_written=true;Token::FarBranch(a,b)
                }
                ["branch_if", target, predicate] => {
                    let a=slot(&names,target)?;let c=slot(&names,predicate)?;
                    if widths[a]!=32 || widths[c]!=1 {return Err("branch_if requires bv<32> target and bv<1> predicate".into());}
                    if branched {return Err("multiple branch effects".into());}
                    if ![16,32].contains(&operand_width) {return Err("branch operand width must be 16 or 32".into());}
                    branched=true;Token::BranchIf(a,c)
                }
                ["read", out, binding, w] => {
                    let w = width(w)?;
                    let b = match *binding { "dst" => Binding::Dst, "src" => Binding::Src,
                        _ => return Err("expected dst or src".into()) };
                    match b {
                        Binding::Dst if w!=operand_width=>return Err("destination binding width mismatch".into()),
                        Binding::Src=>{
                            if ![8,16,32].contains(&w){return Err("source binding width must be 8,16,32".into());}
                            if source_width.is_some_and(|old|old!=w){return Err("inconsistent source binding widths".into());}
                            source_width=Some(w);
                        },_=>{},
                    }
                    Token::Read(define(&mut names, &mut widths, out, w)?, b, w)
                }
                ["const", out, w, n] => {
                    let w = width(w)?; let n = number(n)?;
                    if n & !mask(w) != 0 { return Err("constant does not fit".into()); }
                    Token::Const(define(&mut names, &mut widths, out, w)?, value(n,w))
                }
                ["read_flag", out, name] => {
                    let f = parse_flag(name)?;
                    Token::ReadFlag(define(&mut names, &mut widths, out, 1)?,f)
                }
                ["zext", out, input, w] | ["sext", out, input, w] | ["trunc", out, input, w] => {
                    let a = slot(&names, input)?; let w = width(w)?;
                    let extend = fields[0] != "trunc";
                    if (extend && w < widths[a]) || (!extend && w > widths[a]) {
                        return Err("invalid conversion direction".into());
                    }
                    let o = define(&mut names, &mut widths, out, w)?;
                    if fields[0]=="sext" {Token::SignExtend(o,a,w)} else if extend { Token::Extend(o,a,w) } else { Token::Truncate(o,a,w) }
                }
                [op @ ("add" | "sub" | "mul" | "xor" | "and" | "or"), out, left, right] => {
                    let a = slot(&names,left)?; let b = slot(&names,right)?;
                    if widths[a] != widths[b] { return Err("binary width mismatch".into()); }
                    let w = widths[a];
                    let o = define(&mut names, &mut widths, out, w)?;
                    let op = match *op { "add" => Binary::Add, "sub" => Binary::Sub,
                        "mul" => Binary::Mul, "xor" => Binary::Xor, "or" => Binary::Or, _ => Binary::And };
                    Token::Binary(o,op,a,b)
                }
                ["not", out, input] => {
                    let a = slot(&names,input)?; let w = widths[a];
                    Token::Not(define(&mut names, &mut widths, out, w)?,a)
                }
                ["bit", out, input, index] => {
                    let a = slot(&names,input)?; let i = number(index)?;
                    if i >= widths[a] as u64 { return Err("bit index out of range".into()); }
                    Token::Bit(define(&mut names, &mut widths, out, 1)?,a,i as u8)
                }
                ["zero", out, input] | ["parity", out, input] => {
                    let a = slot(&names,input)?;
                    if fields[0] == "parity" && widths[a] < 8 { return Err("parity requires >=8 bits".into()); }
                    let o = define(&mut names, &mut widths, out, 1)?;
                    if fields[0] == "zero" { Token::Zero(o,a) } else { Token::Parity(o,a) }
                }
                ["write", "dst", input] => {
                    let a = slot(&names,input)?;
                    if widths[a] != operand_width { return Err("write width mismatch".into()); }
                    if wrote { return Err("multiple destination writes".into()); }
                    wrote = true; Token::Write(a)
                }
                ["flag", name, input] => {
                    let a = slot(&names,input)?;
                    if widths[a] != 1 { return Err("flag needs bv<1>".into()); }
                    let f = parse_flag(name)?;
                    if flags_seen & (1 << f.bit()) != 0 { return Err("duplicate flag write".into()); }
                    flags_seen |= 1 << f.bit(); Token::Flag(f,a)
                }
                ["retire"] => { retired = true; Token::Retire }
                _ => return Err("unknown token or wrong operand count".into()),
            }; Ok(t)
        })();
        tokens.push(parsed.map_err(|e| format!("line {}: {e}",line_no+1))?);
    }
    if tokens.iter().any(|t|matches!(t,Token::WriteGpr(..))) && (wrote||wrote_src) {
        return Err("named register effects cannot mix with bound writes".into());
    }
    if !retired { return Err("missing retire".into()); }
    // DF is optional in descriptions; unmentioned non-arithmetic flags stay intact.
    if flags_seen & ARITH_FLAGS != ARITH_FLAGS { return Err("classify all six arithmetic flags: flag, preserve_flag, preserve_flags or undef_flag".into()); }
    let uses_flow=tokens.iter().any(|t| matches!(t,Token::ReadMsw(..)|Token::CheckSegment(..)|Token::CheckpointFlags|Token::RequireRealMode|Token::InterruptShadowIf(..)|Token::ReadIp(..)|Token::ReadDisp(..)|Token::BranchIf(..)|Token::FarBranch(..)|Token::ResumeIf(..)));
    let uses_displacement=tokens.iter().any(|t| matches!(t,Token::ReadDisp(..)));
    let uses_memory=tokens.iter().any(|t|matches!(t,Token::ReadAddressAfter(..)|Token::ReadAddress(..)|Token::Load(..)|Token::Store(..)|Token::LoadSegment(..)|Token::StoreSegment(..)));
    let partial=tokens.iter().any(|t|matches!(t,Token::ResumeIf(..)));
    let checkpoint_flags=tokens.iter().any(|t|matches!(t,Token::CheckpointFlags));
    Ok(Program { checkpoint_flags,partial,tokens, widths, operand_width, source_width:source_width.unwrap_or(operand_width), uses_flow, uses_displacement, uses_memory })
}

#[derive(Clone, Copy, Debug)]
struct View { register: usize, low: u8, width: u8 }
impl View {
    fn validate(self, w: u8) -> R<()> {
        if self.register >= 8 || self.width != w || self.low as u16 + w as u16 > 32 {
            return Err("invalid register binding".into());
        } Ok(())
    }
}
#[derive(Clone, Debug, PartialEq, Eq, Default)]
struct Cpu { regs: [u32;8], eflags: u32, undefined_flags: u32, eip: u32 }
impl Cpu {
    fn read(&self, v: View) -> Value { value((self.regs[v.register] as u64) >> v.low,v.width) }
    fn write(&mut self, v: View, x: Value) {
        let m = mask(v.width) << v.low;
        self.regs[v.register] = (((self.regs[v.register] as u64) & !m) | ((x.bits << v.low) & m)) as u32;
    }
}
fn execute(p: &Program, cpu: &mut Cpu, dst: View, src: View) -> R<()> {
    if p.uses_flow {return Err("flow context required: use execute_flow".into());}
    if p.uses_memory {return Err("RAM context required: use execute_memory".into());}
    let context=FlowContext{next_eip:cpu.eip,displacement:0,displacement_bits:32,code_limit:u32::MAX};
    execute_flow(p,cpu,dst,src,context).map_err(|e|format!("{e:?}"))
}
fn execute_flow(p:&Program,cpu:&mut Cpu,dst:View,src:View,context:FlowContext)->Result<(),ExecError> {
    if p.uses_memory {return Err(ExecError::InvalidContext);}
    execute_state(p,cpu,dst,src,context,None,0,None,None,None)
}
fn execute_memory(p:&Program,cpu:&mut Cpu,dst:View,src:View,context:FlowContext,ram:&mut Ram,address:u32)->Result<(),ExecError> {
    execute_state(p,cpu,dst,src,context,Some(ram),address,None,None,None)
}
#[cfg(test)]
fn execute_segmented(p:&Program,cpu:&mut Cpu,dst:View,src:View,context:FlowContext,ram:&mut Ram,address:address::AddressContext)->Result<(),ExecError> {
    let (offset, memory_context)=address.resolve(&cpu.regs)?;
    execute_state(p,cpu,dst,src,context,Some(ram),offset,Some(memory_context),None,Some(address.effective))
}
fn execute_machine(p:&Program,cpu:&mut Cpu,dst:View,src:View,context:FlowContext,ram:&mut Ram,mut address:Option<address::AddressContext>,machine:&mut address::SegmentState)->Result<(),ExecError>{
    let (offset,memory)=if let Some(a)=address.as_mut(){a.segments=machine.segments;let (o,c)=a.resolve(&cpu.regs)?;(o,Some(c))}else{(0,None)};
    if p.checkpoint_flags{machine.checkpoint_flags(cpu.eip,cpu.eflags,cpu.undefined_flags);}
    let result=execute_state(p,cpu,dst,src,context,Some(ram),offset,memory,Some(&mut *machine),address.map(|a|a.effective));
    if result.is_err()&&p.checkpoint_flags{if let Some((f,u))=machine.recover_flags(){cpu.eflags=f;cpu.undefined_flags=u;}}
    result
}
fn access_error(eip:u32,f:memory::AccessFault)->ExecError {
    ExecError::MemoryAccess{eip,address:f.address,bytes:f.bytes as u8,write:f.write}
}
fn execute_state(p:&Program,cpu:&mut Cpu,dst:View,src:View,context:FlowContext,mut ram:Option<&mut Ram>,address:u32,memory_context:Option<address::MemoryContext>,mut machine:Option<&mut address::SegmentState>,effective:Option<address::EffectiveAddress>)->Result<(),ExecError> {
    if p.uses_memory && ram.is_none() {return Err(ExecError::InvalidContext);}
    dst.validate(p.operand_width).map_err(|_|ExecError::InvalidBinding)?;
    src.validate(p.source_width).map_err(|_|ExecError::InvalidBinding)?;
    context.validate(p.operand_width,p.uses_displacement)?;
    let mut pending_eip=context.next_eip;
    let mut pending_resume=false;
    let mut pending_interrupt_shadow=false;
    let mut pending_segment:Option<(usize,u16,address::SegmentCache)>=None;
    let mut pending_stores:[Option<(u32,MemWidth,u32)>;2]=[None;2];let mut store_index=0;
    let mut values: Vec<Value> = p.widths.iter().map(|&w| value(0,w)).collect();
    let mut pending_dst = None;
    let mut pending_src = None;
    let mut pending_registers:Vec<(View,Value)>=Vec::new();
    let mut pending_flags = cpu.eflags;
    let mut pending_undefined = cpu.undefined_flags;
    for t in &p.tokens {
        match *t {
            Token::CheckpointFlags=>{machine.as_deref().ok_or(ExecError::InvalidContext)?;},
            Token::ReadFlags(o)=>values[o]=value(cpu.eflags as u64,32),
            Token::RequireRealMode=>if !machine.as_deref().ok_or(ExecError::InvalidContext)?.real_mode||cpu.eflags&0x20000!=0{return Err(ExecError::InvalidContext);},
            Token::InterruptShadowIf(c)=>{machine.as_deref().ok_or(ExecError::InvalidContext)?;pending_interrupt_shadow=values[c].bits!=0;},
            Token::Read(o,b,w) => { let v = match b { Binding::Dst => dst, Binding::Src => src };
                values[o] = value(cpu.read(v).bits,w); }
            Token::LoadSegment(o,a,w,segment)=>{
                let m=machine.as_deref().ok_or(ExecError::InvalidContext)?;
                let c=address::MemoryContext{segment:[address::SegmentRegister::Es,address::SegmentRegister::Cs,address::SegmentRegister::Ss,address::SegmentRegister::Ds,address::SegmentRegister::Fs,address::SegmentRegister::Gs][segment],cache:m.segments[segment]};
                let at=c.translate(values[a].bits as u32,w,false,cpu.eip)?;
                values[o]=value(ram.as_deref().ok_or(ExecError::InvalidContext)?.load(at,w).map_err(|f|access_error(cpu.eip,f))? as u64,w.bits());
            }
            Token::StoreSegment(a,v,w,segment)=>{
                let m=machine.as_deref().ok_or(ExecError::InvalidContext)?;
                let c=address::MemoryContext{segment:[address::SegmentRegister::Es,address::SegmentRegister::Cs,address::SegmentRegister::Ss,address::SegmentRegister::Ds,address::SegmentRegister::Fs,address::SegmentRegister::Gs][segment],cache:m.segments[segment]};
                let at=c.translate(values[a].bits as u32,w,true,cpu.eip)?;
                ram.as_deref().ok_or(ExecError::InvalidContext)?.check_write(at,w).map_err(|f|access_error(cpu.eip,f))?;pending_stores[store_index]=Some((at,w,values[v].bits as u32));store_index+=1;
            }
            Token::RetireIf(c)=>if values[c].bits!=0{cpu.eip=context.next_eip;if let Some(m)=machine.as_deref_mut(){m.retired();}return Ok(());},
            Token::ResumeIf(c)=>if values[c].bits!=0{pending_eip=cpu.eip;pending_resume=true;},
            Token::ReadSegment(o,reg)=>values[o]=value(machine.as_deref().ok_or(ExecError::InvalidContext)?.selectors[reg] as u64,16),
            Token::WriteSegment(reg,a)=>{
                let m=machine.as_deref().ok_or(ExecError::InvalidContext)?;
                if !m.real_mode{return Err(ExecError::InvalidContext);}
                let selector=values[a].bits as u16;pending_segment=Some((reg,selector,address::SegmentCache::real_mode(selector)));
            }
            Token::ReadGpr(o,reg,w)=>values[o]=value(cpu.regs[reg] as u64,w),
            Token::WriteGpr(reg,a,w)=>pending_registers.push((View{register:reg,low:0,width:w},values[a])),
            Token::ReadMsw(o)=>values[o]=value(machine.as_deref().ok_or(ExecError::InvalidContext)?.machine_status() as u64,16),
            Token::Require(c)=>if values[c].bits==0{return Err(ExecError::InvalidContext);},
            Token::ReadAddressAfter(o,reg,a)=>{
                let mut regs=cpu.regs;regs[reg]=values[a].bits as u32;
                let ea=effective.ok_or(ExecError::InvalidContext)?;
                values[o]=value(ea.resolve(&regs,None)?.offset as u64,32);
            }
            Token::ReadAddress(o)=>values[o]=value(address as u64,32),
            Token::Load(o,a,w)=>{
                let m=ram.as_deref_mut().ok_or(ExecError::InvalidContext)?;
                let offset=values[a].bits as u32;
                let addr=match memory_context {Some(c)=>c.translate(offset,w,false,cpu.eip)?,None=>offset};
                let v=m.load(addr,w).map_err(|f|access_error(cpu.eip,f))?;
                values[o]=value(v as u64,w.bits());
            }
            Token::Store(a,v,w)=>{
                let m=ram.as_deref_mut().ok_or(ExecError::InvalidContext)?;
                let offset=values[a].bits as u32;
                let addr=match memory_context {Some(c)=>c.translate(offset,w,true,cpu.eip)?,None=>offset};
                m.check_write(addr,w).map_err(|f|access_error(cpu.eip,f))?;
                pending_stores[store_index]=Some((addr,w,values[v].bits as u32));store_index+=1;
            }
            Token::ReadIp(o,next,w) => values[o]=value((if next {context.next_eip} else {cpu.eip}) as u64,w),
            Token::ReadDisp(o,w) => values[o]=value(context.displacement as u32 as u64,w),
            Token::CheckSegment(reg,a,n)=>{
                let m=machine.as_deref().ok_or(ExecError::InvalidContext)?;
                let c=address::MemoryContext{segment:[address::SegmentRegister::Es,address::SegmentRegister::Cs,address::SegmentRegister::Ss,address::SegmentRegister::Ds,address::SegmentRegister::Fs,address::SegmentRegister::Gs][reg],cache:m.segments[reg]};
                c.translate_span(values[a].bits,n,false,cpu.eip)?;
            }
            Token::FarBranch(a,b)=>{
                let m=machine.as_deref().ok_or(ExecError::InvalidContext)?;
                if !m.real_mode||cpu.eflags&0x20000!=0{return Err(ExecError::InvalidContext);}
                let selector=values[a].bits as u16;let cache=address::SegmentCache::real_mode(selector);let target=values[b].bits as u32;
                if target>cache.limit{return Err(ExecError::GeneralProtection{eip:cpu.eip,error_code:0});}
                pending_segment=Some((1,selector,cache));pending_eip=target;
            }
            Token::BranchIf(a,c) => {
                if values[c].bits!=0 {
                    let target=values[a].bits as u32;
                    if target>context.code_limit {return Err(ExecError::GeneralProtection{eip:cpu.eip,error_code:0});}
                    pending_eip=target;
                }
            }
            Token::PreserveFlags(_mask) => {},
            Token::UndefFlag(f) => {
                pending_flags &= !(1u32<<f.bit()); // deterministic zero policy, still undefined
                pending_undefined |= 1u32<<f.bit();
            }
            Token::Select(o,c,a,b) => values[o]=if values[c].bits!=0 {values[a]} else {values[b]},
            Token::WriteIf(a,c) => {if values[c].bits!=0 {pending_dst=Some(values[a]);}},
            Token::ReadFlag(o,f) => values[o] = value(((cpu.eflags >> f.bit()) & 1) as u64,1),
            Token::Const(o,x) => values[o] = x,
            Token::Extend(o,a,w) | Token::Truncate(o,a,w) => values[o] = value(values[a].bits,w),
            Token::SignExtend(o,a,w) => {
                let x=values[a];let bits=if x.bits&(1u64<<(x.width-1))!=0{x.bits|!mask(x.width)}else{x.bits};
                values[o]=value(bits,w);
            }
            Token::Binary(o,op,a,b) => {
                let x = values[a]; let y = values[b];
                let n = match op { Binary::Add => x.bits.wrapping_add(y.bits),
                    Binary::Sub => x.bits.wrapping_sub(y.bits),
                    Binary::Mul => x.bits.wrapping_mul(y.bits),
                    Binary::Xor => x.bits ^ y.bits, Binary::And => x.bits & y.bits, Binary::Or => x.bits | y.bits };
                values[o] = value(n,x.width);
            }
            Token::Shift(o,kind,a,b)=>{
                let x=values[a];let c=values[b].bits;
                let n=match kind {
                    Shift::Left=>if c>=u64::from(x.width){0}else{x.bits<<c},
                    Shift::Right=>if c>=u64::from(x.width){0}else{x.bits>>c},
                    Shift::Arithmetic=>{let signed=((x.bits<<(64-x.width)) as i64)>>(64-x.width);(signed>>c.min(u64::from(x.width-1))) as u64},
                };values[o]=value(n,x.width);
            }
            Token::Less(o,a,b)=>values[o]=value(u64::from(values[a].bits<values[b].bits),1),
            Token::FlagIf(f,a,d,c)=>if values[c].bits!=0{
                let m=1u32<<f.bit();pending_flags&=!m;
                if values[d].bits!=0{pending_flags|=(values[a].bits as u32)<<f.bit();pending_undefined&=!m;}else{pending_undefined|=m;}
            },
            Token::Not(o,a) => values[o] = value(!values[a].bits,values[a].width),
            Token::Bit(o,a,i) => values[o] = value((values[a].bits >> i) & 1,1),
            Token::Zero(o,a) => values[o] = value(u64::from(values[a].bits == 0),1),
            Token::Parity(o,a) => values[o] = value(u64::from((values[a].bits as u8).count_ones() % 2 == 0),1),
            Token::Write(a) => pending_dst = Some(values[a]),
            Token::WriteSource(a)=>pending_src=Some(values[a]),
            Token::WriteSourceIf(a,c)=>if values[c].bits!=0{pending_src=Some(values[a]);},
            Token::Flag(f,a) => { let m = 1u32 << f.bit();
                pending_undefined &= !m;
                pending_flags = (pending_flags & !m) | ((values[a].bits as u32) << f.bit()); }
            Token::Retire => {
                if let(Some(a),Some(b))=(pending_dst,pending_src){
                    let overlap=(mask(dst.width)<<dst.low)&(mask(src.width)<<src.low);
                    if dst.register==src.register&&((a.bits<<dst.low)^(b.bits<<src.low))&overlap!=0{
                        return Err(ExecError::InvalidBinding);
                    }
                }
                for (addr,w,v) in pending_stores.into_iter().flatten() {
                    ram.as_deref_mut().ok_or(ExecError::InvalidContext)?.store(addr,w,v).map_err(|f|access_error(cpu.eip,f))?;
                }
                if let Some(x) = pending_dst { cpu.write(dst,x); }
                if let Some(x) = pending_src { cpu.write(src,x); }
                for (v,x) in pending_registers {cpu.write(v,x);}
                cpu.eflags = pending_flags;
                cpu.undefined_flags = pending_undefined;
                cpu.eip = pending_eip;
                if let Some(m)=machine.as_deref_mut(){
                    if !pending_resume{m.retired();}if pending_interrupt_shadow{m.sti_shadow=1;}if let Some((reg,selector,cache))=pending_segment{m.selectors[reg]=selector;m.segments[reg]=cache;if reg==2{m.ss_shadow=1;}}
                }
                return Ok(());
            }
        }
    }
    Err(ExecError::InvalidContext)
}

fn repeat_description(body:&str,a:u8)->String{
    let body=body.replace("$A",&a.to_string());
    format!("read_gpr count 1 {a}\nzero empty count\nretire_if empty\n{}\nconst one {a} 1\nsub remaining count one\nwrite_gpr 1 remaining\nzero done remaining\nnot again done\nresume_if again\npreserve_flags all\nretire",body.strip_suffix("preserve_flags all\nretire\n").unwrap())
}
fn compare_repeat_description(body:&str,a:u8,equal:bool)->String{
    let body=body.replace("$A",&a.to_string());let tail=if equal{"and again more zf"}else{"not different zf\nand again more different"};
    format!("checkpoint_flags\nread_gpr count 1 {a}\nzero empty count\nretire_if empty\n{}\nconst one {a} 1\nsub remaining count one\nwrite_gpr 1 remaining\nzero done remaining\nnot more done\n{tail}\nresume_if again\nretire",body.strip_suffix("retire\n").unwrap())
}

fn instantiate(text: impl AsRef<str>, w: u8) -> String {
    text.as_ref().replace("$WRAP_GUARD","").replace("$HIGH_MASK","4294901760").replace("$FAR_TARGET","const target_value $W 0\nconst selector 16 0").replace("$TARGET","read_disp delta $W\nadd target_value return_offset delta").replace("$CLEANUP","0").replace("$FRAME",&(w/4).to_string()).replace("$SEG","3").replace("$A","16").replace("$S\n","16\n").replace("$S ","16 ").replace("$BYTES",&(w/8).to_string()).replace("$REG","0").replace("$DOUBLE",&(2*w as u16).to_string()).replace("$EXT",&(w+1).to_string()).replace("$SIGN",&(w-1).to_string()).replace("$W",&w.to_string())
}
fn emit_rust(p: &Program) -> String {
    let mut out=String::from("// Generated from validated PIL tokens.\n");
    out.push_str(include_str!("flow.rs"));
    // The included module has an inner documentation comment; in the emitted
    // standalone body it must be an ordinary comment.
    out.push_str("\n");out.push_str(include_str!("memory.rs").split("#[cfg(test)]").next().unwrap());
    out.push_str("\npub mod address {\n");
    out.push_str(include_str!("address.rs").split("#[cfg(test)]").next().unwrap());
    out.push_str("\n}\n");
    out=out.lines().map(|line|line.strip_prefix("//!").map(|s|format!("//{s}")).unwrap_or_else(||line.to_owned())).collect::<Vec<_>>().join("\n");
    out.push_str(&emit_rust_functions(p));out
}
// Shared support types can be supplied by a compiled guest runner instead of
// duplicated in each generated instruction module.
fn emit_rust_functions(p:&Program)->String {
    let mut out=String::new();
    out.push_str(&format!(
        "\nfn execute_state(regs: &mut [u32;8], flags: &mut u32, undefined: &mut u32, eip: &mut u32, dst: (usize,u8), src: (usize,u8), context: FlowContext, mut ram: Option<&mut Ram>, address: u32, memory_context: Option<address::MemoryContext>, mut machine: Option<&mut address::SegmentState>, effective: Option<address::EffectiveAddress>) -> Result<(), ExecError> {{\n\
         let w = {}u8;\n\
         if dst.0 >= 8 || src.0 >= 8 || dst.1 as u16 + w as u16 > 32 || src.1 as u16 + {}u16 > 32 {{ return Err(ExecError::InvalidBinding); }}\n\
         context.validate(w,{})?;\n\
         if {} && ram.is_none() {{ return Err(ExecError::InvalidContext); }}\n\
         let mut pending_segment: Option<(usize,u16,address::SegmentCache)> = None;\n\
         let mut pending_stores: [Option<(u32,MemWidth,u32)>;2] = [None;2];let mut store_index=0;\n\
         let mut pending_eip = context.next_eip;\n\
         let mut pending_resume = false;\nlet mut pending_interrupt_shadow = false;\n\

    fn emit(&mut self,data:&[u8]){self.bytes.extend_from_slice(data);}
    fn label(&mut self,name:&'static str){self.labels.insert(name,self.bytes.len());}
    fn imm(&mut self,opcode:&[u8],value:i32){self.emit(opcode);self.emit(&value.to_le_bytes());}
    fn jump(&mut self,cc:Option<u8>,name:&'static str){if let Some(cc)=cc{self.emit(&[0x0f,0x80+cc]);}else{self.emit(&[0xe9]);}self.fixups.push((self.bytes.len(),name));self.emit(&[0,0]);}
    fn finish(mut self)->Result<Vec<u8>,String>{for(at,name)in self.fixups{let target=*self.labels.get(name).ok_or("unknown assembly label")?;let delta=i16::try_from(target as i64-(at+2) as i64).map_err(|_|"branch out of range")?;self.bytes[at..at+2].copy_from_slice(&delta.to_le_bytes());}Ok(self.bytes)}
}

//! Flat RAM only: no MMIO, segmentation, page tables or guest #PF delivery.
#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum MemWidth { Byte, Word, Dword }
impl MemWidth {
    pub fn from_bits(bits:u8)->Option<Self> {
        match bits {8=>Some(Self::Byte),16=>Some(Self::Word),32=>Some(Self::Dword),_=>None}
    }
    pub fn bits(self)->u8 {self.bytes()*8}
    pub fn bytes(self)->u8 {match self {Self::Byte=>1,Self::Word=>2,Self::Dword=>4}}
}
#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct AccessFault {pub address:u32,pub bytes:usize,pub write:bool}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Ram {
    data:Vec<u8>,
    readonly:Vec<std::ops::Range<usize>>,
}
impl Ram {
    pub fn new(bytes:usize)->Self {Self{data:vec![0;bytes],readonly:Vec::new()}}
    pub fn bytes(&self)->&[u8] {&self.data}
    fn range(&self,address:u32,bytes:usize,write:bool)->Result<std::ops::Range<usize>,AccessFault> {
        let fault=AccessFault{address,bytes,write};
        let start=usize::try_from(address).map_err(|_|fault)?;
        let end=start.checked_add(bytes).ok_or(fault)?;
        if end>self.data.len() {return Err(fault);}
        // A 32-bit guest access cannot wrap to the beginning of RAM.
        if (address as u64)+(bytes as u64)>0x1_0000_0000 {return Err(fault);}
        if write && bytes!=0 && self.readonly.iter().any(|r|start<r.end && end>r.start) {return Err(fault);}
        Ok(start..end)
    }
    pub fn protect(&mut self,address:u32,bytes:usize)->Result<(),AccessFault> {
        let range=self.range(address,bytes,false)?;
        if !range.is_empty() {self.readonly.push(range);}
        Ok(())
    }
    pub fn check_write(&self,address:u32,width:MemWidth)->Result<(),AccessFault> {
        self.range(address,width.bytes() as usize,true).map(|_|())
    }
    pub fn load(&self,address:u32,width:MemWidth)->Result<u32,AccessFault> {
        let range=self.range(address,width.bytes() as usize,false)?;
        let mut value=0u32;
        for (shift,byte) in self.data[range].iter().enumerate() {value|=u32::from(*byte)<<(shift*8);}
        Ok(value)
    }
    pub fn store(&mut self,address:u32,width:MemWidth,value:u32)->Result<(),AccessFault> {
        let range=self.range(address,width.bytes() as usize,true)?;
        for (shift,index) in range.enumerate() {self.data[index]=(value>>(shift*8)) as u8;}
        Ok(())
    }
    // Debugger bulk writes have the same full-range permission check.
    pub fn write_bytes(&mut self,address:u32,data:&[u8])->Result<(),AccessFault> {
        let range=self.range(address,data.len(),true)?;
        self.data[range].copy_from_slice(data);Ok(())
    }
}

fn emit_case(case:&Case,name:&str)->Result<String,String> {
    let (_,ram)=Image::load(&case.bytes)?;let mut ip=guest::ENTRY;
    let mut code=String::new();let mut dispatch=String::new();
    while ip<case.image.end {
        let decoded=case.image.decode(&ram,ip).map_err(|e|format!("{} decode: {e:?}",case.name))?;
        let signature=case.image.fetch_signature(&ram,ip,decoded.length).map_err(|e|format!("{e:?}"))?;
        let bytes=&signature[..decoded.length as usize];
        let physical=case.image.segments[1].base.wrapping_add(ip);
        dispatch.push_str(&format!("{physical}=>{{if let Err(status)=guard(s,s.eip,&{bytes:?},s.machine.segments[1],{}){{return status;}}\n",case.image.end));
        match decoded.operation {
            Operation::Pil{program,dst,src,flow,address}=>{
                code.push_str(&format!("mod i{ip:x} {{use super::{{FlowContext,ExecError,Ram,MemWidth,address}}; {} }}\n",super::emit_rust_functions(&program)));
                let dynamic_flow=format!("FlowContext{{next_eip:s.eip+{},code_limit:s.machine.segments[1].limit,..{flow:?}}}",decoded.length);
                dispatch.push_str(&format!("let flow_context={dynamic_flow};\n"));
                let args=format!("&mut s.regs,&mut s.flags,&mut s.undefined,&mut s.eip,({},{}),({},{}),flow_context",dst.register,dst.low,src.register,src.low);
                let mut call=format!("i{ip:x}::execute_with_machine({args},&mut s.ram,{address:?},&mut s.machine)");
                if program.tokens.iter().any(|t|matches!(t,super::Token::ReadGpr(_,4,_)|super::Token::WriteGpr(4,_,_))){
                    let mut alternate=case.image.clone();let old_big=matches!(alternate.segments[2].access,super::address::SegmentAccess::Data{big:true,..});
                    if let super::address::SegmentAccess::Data{ref mut big,..}=alternate.segments[2].access{*big=!old_big;}
                    if let Operation::Pil{program:p,dst:d,src:r,flow:_,address:a}=alternate.decode(&ram,ip).map_err(|e|format!("alternate stack decode: {e:?}"))?.operation{
                        code.push_str(&format!("mod alt{ip:x} {{use super::{{FlowContext,ExecError,Ram,MemWidth,address}}; {} }}\n",super::emit_rust_functions(&p)));
                        let alternate_call=format!("alt{ip:x}::execute_with_machine(&mut s.regs,&mut s.flags,&mut s.undefined,&mut s.eip,({},{}),({},{}),flow_context,&mut s.ram,{a:?},&mut s.machine)",d.register,d.low,r.register,r.low);
                        call=format!("if matches!(s.machine.segments[2].access,SegmentAccess::Data{{big:true,..}})=={old_big}{{{call}}}else{{{alternate_call}}}");
                    }
                }
                dispatch.push_str(&format!("match {call} {{Ok(())=>[0,0,0,0,0],Err(e)=>trace::fault(s.eip,e)}}"));
            }
            Operation::Input{next}=>{let delta=next-ip;dispatch.push_str(&format!("if let Some(b)=s.rx.pop_front(){{s.regs[0]=(s.regs[0]&!255)|u32::from(b);s.eip=s.eip+{delta};s.machine.retired();[0,0,0,0,0]}}else{{[2,s.eip,0,0,0]}}"));},
            Operation::Output{next}=>{let delta=next-ip;dispatch.push_str(&format!("if s.tx.len()==256{{[3,s.eip,0,0,0]}}else{{s.tx.push_back(s.regs[0] as u8);s.eip=s.eip+{delta};s.machine.retired();[0,0,0,0,0]}}"));},
            Operation::Halt=>dispatch.push_str("[1,s.eip,0,0,0]"),
        }
        dispatch.push_str("},\n");ip+=decoded.length;
    }
    let mut init=format!("let mut ram=Ram::new({});ram.write_bytes(0x100,&{:?}).unwrap();\n",guest::RAM_BYTES,case.bytes);
    if let Some((at,bytes))=&case.patch {init.push_str(&format!("ram.write_bytes({at},&{bytes:?}).unwrap();\n"));}
    if let Some((at,len))=case.protect {init.push_str(&format!("ram.protect({at},{len}).unwrap();\n"));}
    init.push_str(&format!("State{{regs:{:?},flags:{},undefined:0,eip:0x100,ram,machine:{:?},rx:VecDeque::from({:?}),tx:VecDeque::from({:?})}}",case.regs,case.flags,case.image.segment_state(),case.rx,case.tx));
    Ok(format!("mod {name} {{use super::*; const CS:SegmentCache={:?};\n{code}\npub fn initial()->State{{{init}}}\n#[inline(never)] pub fn step(s:&mut State)->[u32;5]{{match s.machine.segments[1].base.wrapping_add(s.eip){{{dispatch}_=>[7,s.eip,0,0,0]}}}}\n}}\n",case.image.segments[SegmentRegister::Cs.index()]))
}
fn runner_source()->String {
    let mut source=String::from("// Generated from validated PIL; fixed-image native Rust proof, not a JIT.\n#![allow(dead_code,unused_variables,unused_mut,unused_imports)]\nuse std::collections::VecDeque;\n");
    source.push_str("mod flow {\n");source.push_str(include_str!("flow.rs"));source.push_str("\n}\nuse flow::{ExecError,FlowContext};\n");
    source.push_str("mod memory {\n");source.push_str(include_str!("memory.rs").split("#[cfg(test)]").next().unwrap());source.push_str("\n}\nuse memory::{Ram,MemWidth};\n");
    source.push_str("mod address {\n");source.push_str(include_str!("address.rs").split("#[cfg(test)]").next().unwrap());source.push_str("\n}\nuse address::*;\nuse address::{EffectiveAddress::*,Address16::*,Gpr::*,SegmentRegister::*,SegmentAccess::*};\n");
    source.push_str("mod trace {\n");source.push_str(include_str!("trace.rs"));source.push_str("\n}\n");

}
pub fn emit_suite(iterations:u32,samples:u32)->Result<String,String> {
    validate(iterations,samples)?;let tests=cases()?;
    let mut source=runner_source();
    for (id,test) in tests.iter().enumerate(){source.push_str(&emit_case(test,&format!("c{id}"))?);}
    source.push_str(&emit_case(&case("timed_loop",guest::loop_bytes(iterations)?)?,"timed_loop")?);

    for (id,test) in tests.iter().enumerate(){source.push_str(&format!("let (state,done,status)=run(c{id}::initial(),c{id}::step,1000,Some((&mut trace_bytes,{id})));assert_ne!(status[0],10);println!(\"GENERATED verify {}: steps={{done}} stop={{}}\",status[0]);\n",test.name));}
    source.push_str("std::fs::write(path,trace_bytes).unwrap();}if mode==\"verify\"{return;}\nlet mut samples=Vec::new();\n");

    Ok(source)

// Parse borrowed slices: damaged length fields cannot cause allocations or panics.
struct TraceRecord<'a> { header:[u32;17], sections:[&'a [u8];4] }
fn trace_record<'a>(bytes:&'a [u8],at:&mut usize)->Result<Option<TraceRecord<'a>>,String> {
    if *at==bytes.len(){return Ok(None);}
    fn take<'a>(bytes:&'a [u8],at:&mut usize,len:usize)->Result<&'a [u8],String> {
        let end=at.checked_add(len).ok_or("trace length overflow")?;
        let part=bytes.get(*at..end).ok_or_else(||format!("truncated trace at byte {}: need {len}, {} remain",*at,bytes.len().saturating_sub(*at)))?;
        *at=end;Ok(part)
    }
    let mut header=[0;17];
    for value in &mut header {*value=u32::from_le_bytes(take(bytes,at,4)?.try_into().unwrap());}
    let mut sections:[&[u8];4]=[&[];4];
    for section in &mut sections {let len=u32::from_le_bytes(take(bytes,at,4)?.try_into().unwrap()) as usize;*section=take(bytes,at,len)?;}
    Ok(Some(TraceRecord{header,sections}))
}
pub fn verify_traces(expected:&[u8],actual:&[u8])->Result<(),String> {
    let tests=cases()?;let names:Vec<_>=tests.iter().map(|c|c.name).collect();
    verify_named_traces(expected,actual,&names,"step")
}
fn verify_named_traces(expected:&[u8],actual:&[u8],names:&[&str],unit:&str)->Result<(),String> {
    const FIELDS:[&str;17]=["case","stop.kind","stop.eip","stop.detail0","stop.detail1","stop.detail2","EAX","ECX","EDX","EBX","ESP","EBP","ESI","EDI","EFLAGS","undefined_flags","EIP"];
    let (mut left,mut right,mut record,mut step)=(0,0,0,0);
    let mut previous=None;let mut start_eip=guest::ENTRY;
    loop {
        record+=1;
        let a=trace_record(expected,&mut left).map_err(|e|format!("interpreter record {record}: {e}"))?;
        let b=trace_record(actual,&mut right).map_err(|e|format!("generated record {record}: {e}"))?;
        let (a,b)=match (a,b) {
            (None,None)=>return if record==1{Err("empty backend traces".into())}else{Ok(())},
            (Some(a),Some(b))=>(a,b),
            (a,_)=>return Err(format!("backend trace record {record}: {} record missing",if a.is_none(){"interpreter"}else{"generated"})),
        };
        let id=a.header[0];
        if previous!=Some(id){step=0;start_eip=guest::ENTRY;}
        step+=1;
        let name=names.get(id as usize).copied().unwrap_or("unknown");
        if unit=="checkpoint"{start_eip=a.header[16];}
        let context=format!("backend trace case {id} ({name}), {unit} {step}, record {record}, instruction EIP={start_eip:08x}");
        for (index,field) in FIELDS.iter().enumerate(){
            if a.header[index]!=b.header[index]{return Err(format!("{context}: {field}: interpreter={:08x}, generated={:08x}",a.header[index],b.header[index]));}
        }
        for (index,field) in ["RAM","RX","TX","segments"].iter().enumerate(){
            let (a,b)=(a.sections[index],b.sections[index]);
            if a.len()!=b.len(){return Err(format!("{context}: {field}.length: interpreter={}, generated={}",a.len(),b.len()));}
            if let Some(offset)=a.iter().zip(b).position(|(a,b)|a!=b){return Err(format!("{context}: {field}[0x{offset:x}]: interpreter={:02x}, generated={:02x}",a[offset],b[offset]));}
        }
        previous=Some(id);start_eip=a.header[16];
    }
}
struct Proof {stdout:String,count:usize,bytes:usize,compile_secs:f64,executable:PathBuf,trace_path:PathBuf}
fn proof_at(directory:&Path,iterations:u32,samples:u32)->Result<Proof,String> {
    validate(iterations,samples)?;let tests=cases()?;let source=emit_suite(iterations,samples)?;
    fs::create_dir_all(directory).map_err(|e|e.to_string())?;
    let source_path=directory.join("generated.rs");let executable=directory.join(if cfg!(windows){"generated.exe"}else{"generated"});let trace_path=directory.join("generated.trace");
    fs::write(&source_path,source).map_err(|e|e.to_string())?;
    let start=Instant::now();let rustc=std::env::var_os("RUSTC").unwrap_or_else(||"rustc".into());
    let built=Command::new(rustc).arg("--edition=2021").arg("-O").arg(&source_path).arg("-o").arg(&executable).output().map_err(|e|format!("cannot run rustc: {e}"))?;
    let compile_time=start.elapsed();if !built.status.success(){return Err(format!("generated Rust build failed: {}",String::from_utf8_lossy(&built.stderr)));}
    // Verification is completed before either engine's timed comparison runs.
    let output=Command::new(&executable).arg(&trace_path).arg("verify").output().map_err(|e|e.to_string())?;
    if !output.status.success(){return Err(format!("generated verifier failed: {}",String::from_utf8_lossy(&output.stderr)));}
    let stdout=String::from_utf8(output.stdout).map_err(|e|e.to_string())?;
    let expected=super::debugger::backend_trace(&tests)?;let actual=fs::read(&trace_path).map_err(|e|e.to_string())?;
    // Preserve both sides even on mismatch so the failure is reproducible.
    fs::write(directory.join("interpreter.trace"),&expected).map_err(|e|e.to_string())?;
    verify_traces(&expected,&actual).map_err(|e|format!("{e}\nArtifacts: {}",directory.display()))?;
    Ok(Proof{stdout,count:tests.len(),bytes:expected.len(),compile_secs:compile_time.as_secs_f64(),executable,trace_path})
}

}
pub fn compare_at(directory:&Path,iterations:u32,samples:u32)->Result<String,String> {
    let proof=proof_at(directory,iterations,samples)?;
    let output=Command::new(&proof.executable).arg(&proof.trace_path).arg("bench").output().map_err(|e|e.to_string())?;
    if !output.status.success(){return Err(format!("generated benchmark failed: {}",String::from_utf8_lossy(&output.stderr)));}
    let timed=String::from_utf8(output.stdout).map_err(|e|e.to_string())?;
    let median_ns=timed.lines().find_map(|line|line.strip_prefix("NATIVE_RESULT median_ns=")).and_then(|line|line.split_whitespace().next()).ok_or("missing native timing")?.parse::<u128>().map_err(|e|e.to_string())?;
    let (interpreter,median)=super::debugger::benchmark_details(iterations,samples,true)?;
    let native_ms=median_ns as f64/1e6;let native_ips=if median_ns==0{0.0}else{(3*iterations+3) as f64*1e9/median_ns as f64};
    let ratio=if median_ns==0{0.0}else{median/(median_ns as f64/1e9)};
    let cpu=cpu_benchmark(iterations,samples)?;
    Ok(format!("PASS: generated Rust matches interpreter byte-for-byte after every execution step across {} cases ({} trace bytes).\n{}\n{timed}\n{interpreter}\nGenerated Rust median={native_ms:.3} ms; {native_ips:.0} guest instructions/s; observed ratio={ratio:.2}x vs cached interpreter.\nGenerated build={:.3} s (excluded from samples); rustc -O, fixed-image guarded dispatch, black_box per step. No JIT; debugger remains token-interpreted.\n{cpu}\nArtifacts: {}",proof.count,proof.bytes,proof.stdout,proof.compile_secs,directory.display()))
}

#[cfg(target_arch="x86_64")]
#[inline(never)]
fn cpu_loop(iterations:u32)->(u32,u32) {
    let answer:u32;let remaining:u32;
    // Register-only bounded arithmetic. ESI is input-only; EAX/ECX/EDX and
    // flags are outputs/clobbers. No memory access or stack manipulation.
    unsafe {std::arch::asm!(
        "mov eax, 0", "mov ecx, esi", "mov edx, 1",
        "2:", "add eax, edx", "sub ecx, edx", "jnz 2b",
        in("esi") iterations,lateout("eax") answer,lateout("ecx") remaining,lateout("edx") _,
        options(nomem,nostack)
    );}
    (answer,remaining)
}
pub fn cpu_benchmark(iterations:u32,samples:u32)->Result<String,String> {
    validate(iterations,samples)?;
    #[cfg(target_arch="x86_64")]
    {
        let mut times=Vec::new();
        for sample in 0..=samples {
            let start=Instant::now();let (answer,remaining)=cpu_loop(std::hint::black_box(iterations));let ns=start.elapsed().as_nanos();
            if answer!=iterations || remaining!=0{return Err("native CPU loop result mismatch".into());}
            if sample!=0{times.push(ns);}
        }
        let values=times.iter().map(|n|*n as f64/1e6).collect::<Vec<_>>();times.sort();let mid=times.len()/2;
        let median=if times.len()%2==0{(times[mid-1]+times[mid])/2}else{times[mid]};
        Ok(format!("PASS: PC_NATIVE x86_64 ADD/SUB/JNZ; {iterations} iterations; EAX={iterations}, ECX=0 verified\nsamples_ms={values:?}; median={:.6} ms; {:.3} ns/iteration; {samples} measured samples, 1 excluded warmup.\nEquivalent 32-bit-register arithmetic loop in 64-bit mode; includes call/timer overhead, excludes decoder/PIL/guards. Not execution of the 16-bit image and not 386 MHz.",median as f64/1e6,median as f64/iterations as f64))
    }
    #[cfg(not(target_arch="x86_64"))]
    {Ok(format!("PC_NATIVE unavailable on {}; x86_64 host required. The generated Rust runner still compiles for this host.",std::env::consts::ARCH))}
}

//! Nonblocking byte transport. No UART register or PC port emulation yet.
use std::collections::VecDeque;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum SerialError {
    WouldBlock,
}

pub trait SerialTransport {
    fn rx_ready(&self) -> bool;
    fn try_read(&mut self) -> Option<u8>;
    fn try_write(&mut self, byte: u8) -> Result<(), SerialError>;
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct MockSerial {
    rx: VecDeque<u8>,
    tx: VecDeque<u8>,
    rx_capacity: usize,
    tx_capacity: usize,
}

impl MockSerial {
    pub fn new(rx_capacity: usize, tx_capacity: usize) -> Self {
        Self { rx: VecDeque::new(), tx: VecDeque::new(), rx_capacity, tx_capacity }
    }
    // Host/test side: queue a complete chunk or nothing.
    pub fn inject_rx(&mut self, bytes: &[u8]) -> Result<(), SerialError> {
        if bytes.len() > self.rx_capacity - self.rx.len() {
            return Err(SerialError::WouldBlock);
        }
        self.rx.extend(bytes.iter().copied());
        Ok(())
    }
    // Host/test side: collect guest-transmitted bytes, including binary zero.
    pub fn take_tx(&mut self) -> Vec<u8> { self.tx.drain(..).collect() }
    pub fn snapshot(&self)->(Vec<u8>,Vec<u8>) {(self.rx.iter().copied().collect(),self.tx.iter().copied().collect())}
    pub fn lengths(&self) -> (usize, usize) { (self.rx.len(), self.tx.len()) }
    pub fn capacities(&self) -> (usize, usize) { (self.rx_capacity, self.tx_capacity) }
    pub fn clear(&mut self) { self.rx.clear(); self.tx.clear(); }
}

impl SerialTransport for MockSerial {
    fn rx_ready(&self) -> bool { !self.rx.is_empty() }
    fn try_read(&mut self) -> Option<u8> { self.rx.pop_front() }
    fn try_write(&mut self, byte: u8) -> Result<(), SerialError> {
        if self.tx.len() >= self.tx_capacity { return Err(SerialError::WouldBlock); }
        self.tx.push_back(byte);
        Ok(())
    }
}

pub fn fault(eip:u32,error:ExecError)->[u32;5] {
    match error {
        ExecError::GeneralProtection{eip,error_code}=>[4,eip,error_code as u32,0,0],
        ExecError::StackSegment{eip,error_code}=>[5,eip,error_code as u32,0,0],
        ExecError::MemoryAccess{eip,address,bytes,write}=>[6,eip,address,bytes as u32,u32::from(write)],
        ExecError::InvalidBinding=>[8,eip,0,0,0],ExecError::InvalidContext=>[9,eip,0,0,0],
    }
}
