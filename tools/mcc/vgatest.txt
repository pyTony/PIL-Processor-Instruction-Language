# svgalib's vgatest in 320x200x256 (5) and 640x480x16 (4), after
# svgalib.txt, with a screenshot of each (P6 PPM in the current folder):
#   pil --boot-probe images/mcc-1.0/nocdboot --hd-image images/mcc-1.0/hd40.img
#       --script tools/mcc/vgatest.txt --fpu 80387 --steps 3000000000 --rtc-year 1994
# vgatest waits for Enter in graphics; the runs are fixed step counts.
run 3000000
type ro root=301\r

type root\r
prompt >
type /usr/local/bin/vgatest\r
run 60000000
show
type 5\r
run 150000000
shot vgatest-5.ppm
type \r
run 60000000
type /usr/local/bin/vgatest\r
run 60000000
type 4\r
run 200000000
shot vgatest-4.ppm
type \r
run 60000000
show
